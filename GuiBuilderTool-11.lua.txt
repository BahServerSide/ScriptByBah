local BORDERED_CLASSES = {

local function new(className, props, children)
	local inst = Instance.new(className)
	if BORDERED_CLASSES[className] then
		inst.BorderSizePixel = 0
	end
	for prop, value in pairs(props or {}) do
		inst[prop] = value
	end
	for _, child in ipairs(children or {}) do
		child.Parent = inst
	end
	return inst
end

local function round(number, decimals)
	local mult = 10 ^ (decimals or 0)
	return math.floor(number * mult + 0.5) / mult
end

local function colorToHex(color)
	return string.format("%02X%02X%02X",
		math.floor(color.R * 255 + 0.5),
		math.floor(color.G * 255 + 0.5),
		math.floor(color.B * 255 + 0.5))
end

local function hexToColor(hex)
	hex = hex:gsub("#", ""):gsub("%s", "")
	if #hex ~= 6 then return nil end
	local r = tonumber(hex:sub(1, 2), 16)
	local g = tonumber(hex:sub(3, 4), 16)
	local b = tonumber(hex:sub(5, 6), 16)
	if not (r and g and b) then return nil end
	return Color3.fromRGB(r, g, b)
end

-- Inset correction (0 if source ScreenGui also ignores GuiInset)
local function getInsetCorrection(inst)
	local node = inst
	while node and not node:IsA("ScreenGui") do
		node = node.Parent
	end
	if node and node.IgnoreGuiInset then
		return Vector2.new(0, 0)
	end
	return GuiService:GetGuiInset()
end

-- Drag tool windows
local function makeDraggable(frame, handle)
	local dragging = false
	local dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

local function luaValue(value)
	local t = typeof(value)
	if t == "string" then
		return string.format("%q", value)
	elseif t == "boolean" or t == "number" then
		return tostring(value)
	elseif t == "UDim2" then
		return string.format("UDim2.new(%s, %s, %s, %s)",
			round(value.X.Scale, 4), value.X.Offset, round(value.Y.Scale, 4), value.Y.Offset)
	elseif t == "UDim" then
		return string.format("UDim.new(%s, %s)", round(value.Scale, 4), value.Offset)
	elseif t == "Color3" then
		return string.format("Color3.fromRGB(%d, %d, %d)",
			math.floor(value.R * 255 + 0.5), math.floor(value.G * 255 + 0.5), math.floor(value.B * 255 + 0.5))
	elseif t == "Vector2" then
		return string.format("Vector2.new(%s, %s)", round(value.X, 4), round(value.Y, 4))
	elseif t == "EnumItem" then
		return "Enum." .. tostring(value.EnumType) .. "." .. value.Name
	else
		return tostring(value)
	end
end

local function getParentAbs(parent)

		return parent.AbsolutePosition, parent.AbsoluteSize
	end
	local cam = Workspace.CurrentCamera

local function makeAlignLine(isVertical, pos, length, offset, inset)

				inset = inset or GuiService:GetGuiInset()

		Name = "__AlignLine",

		Parent = alignmentFolder,
	})
	if isVertical then
		line.Size = UDim2.fromOffset(1, length)
		line.Position = UDim2.fromOffset(pos + inset.X, offset + inset.Y)
	else
		line.Size = UDim2.fromOffset(length, 1)
		line.Position = UDim2.fromOffset(offset + inset.X, pos + inset.Y)
	end
	table.insert(alignmentLines, line)
	return line
end

-- Snap against parent and sibling edges
local function snapToGuides(inst, proposedAbsPos, proposedAbsSize)
	if not alignmentEnabled or not inst or not inst:IsA("GuiObject") then
		return proposedAbsPos, proposedAbsSize, {}
	end
	clearAlignmentLines()

	local inset = getInsetCorrection(inst)
	local parent = inst.Parent
	local parentPos, parentSize
	if parent and parent:IsA("GuiObject") then
		parentPos = parent.AbsolutePosition
		parentSize = parent.AbsoluteSize
	else
		local cam = Workspace.CurrentCamera
		local vs = cam and cam.ViewportSize or Vector2.new(1280, 720)
		parentPos = Vector2.new(0, 0)
		parentSize = vs
	end

	local px, py = proposedAbsPos.X, proposedAbsPos.Y
	local w, h = proposedAbsSize.X, proposedAbsSize.Y
	local snappedX, snappedY = px, py
	local guides = {}

	-- Parent reference points
	local refsX = {
		{ pos = parentPos.X, kind = "left" },
		{ pos = parentPos.X + parentSize.X / 2, kind = "center" },
		{ pos = parentPos.X + parentSize.X, kind = "right" },
	}
	local refsY = {
		{ pos = parentPos.Y, kind = "top" },
		{ pos = parentPos.Y + parentSize.Y / 2, kind = "center" },
		{ pos = parentPos.Y + parentSize.Y, kind = "bottom" },
	}

	-- Element's own edges
	local function trySnapX(edgeAbs)
		for _, ref in ipairs(refsX) do
			if math.abs(edgeAbs - ref.pos) <= snapThreshold then
				return ref.pos, ref
			end
		end
		return nil, nil
	end
	local function trySnapY(edgeAbs)
		for _, ref in ipairs(refsY) do
			if math.abs(edgeAbs - ref.pos) <= snapThreshold then
				return ref.pos, ref
			end
		end
		return nil, nil
	end

	-- left
	local s, ref = trySnapX(px)
	if s then
		snappedX = s
		table.insert(guides, { vertical = true, pos = s })
	else
		-- center
		s, ref = trySnapX(px + w / 2)
		if s then
			snappedX = s - w / 2
			table.insert(guides, { vertical = true, pos = s })
		else
			-- right
			s, ref = trySnapX(px + w)
			if s then
				snappedX = s - w
				table.insert(guides, { vertical = true, pos = s })
			end
		end
	end

	s, ref = trySnapY(py)
	if s then
		snappedY = s
		table.insert(guides, { vertical = false, pos = s })
	else
		s, ref = trySnapY(py + h / 2)
		if s then
			snappedY = s - h / 2
			table.insert(guides, { vertical = false, pos = s })
		else
			s, ref = trySnapY(py + h)
			if s then
				snappedY = s - h
				table.insert(guides, { vertical = false, pos = s })
			end
		end
	end

	-- Snap against siblings
	if parent then
		for _, sib in ipairs(parent:GetChildren()) do
			if sib ~= inst and sib:IsA("GuiObject") and not isOverlay(sib) then
				local sx, sy = sib.AbsolutePosition.X, sib.AbsolutePosition.Y
				local sw, sh = sib.AbsoluteSize.X, sib.AbsoluteSize.Y
				local sibRefsX = { sx, sx + sw / 2, sx + sw }
				local sibRefsY = { sy, sy + sh / 2, sy + sh }
				for _, rp in ipairs(sibRefsX) do
					if math.abs(snappedX - rp) <= snapThreshold then
						snappedX = rp
						table.insert(guides, { vertical = true, pos = rp })
					elseif math.abs(snappedX + w / 2 - rp) <= snapThreshold then
						snappedX = rp - w / 2
						table.insert(guides, { vertical = true, pos = rp })
					elseif math.abs(snappedX + w - rp) <= snapThreshold then
						snappedX = rp - w
						table.insert(guides, { vertical = true, pos = rp })
					end
				end
				for _, rp in ipairs(sibRefsY) do
					if math.abs(snappedY - rp) <= snapThreshold then
						snappedY = rp
						table.insert(guides, { vertical = false, pos = rp })
					elseif math.abs(snappedY + h / 2 - rp) <= snapThreshold then
						snappedY = rp - h / 2
						table.insert(guides, { vertical = false, pos = rp })
					elseif math.abs(snappedY + h - rp) <= snapThreshold then
						snappedY = rp - h
						table.insert(guides, { vertical = false, pos = rp })
					end
				end
			end
		end
	end

	-- Draw guide lines
	for _, g in ipairs(guides) do
		if g.vertical then
			makeAlignLine(true, g.pos, parentSize.Y, parentPos.Y, inset)
		else
			makeAlignLine(false, g.pos, parentSize.X, parentPos.X, inset)
		end
	end

	return Vector2.new(snappedX, snappedY), proposedAbsSize, guides
end

local function applySnappedPosition(inst, absPos, absSize)
	local parentPos = getParentAbs(inst.Parent)
	local relX = absPos.X - parentPos.X
	local relY = absPos.Y - parentPos.Y
	inst.Position = UDim2.new(0, relX, 0, relY)
	inst.Size = UDim2.new(0, absSize.X, 0, absSize.Y)
end


function updateBoundingBox()
	if not selected or not selected:IsA("GuiObject") or not boundingBoxFrame then return end
	if not selected.Parent then
		destroyBoundingBox()
		return
	end

	-- Mirror in the SAME parent (avoids GuiInset teleport)
	if boundingBoxFrame.Parent ~= selected.Parent then
		boundingBoxFrame.Parent = selected.Parent
	end
	boundingBoxFrame.AnchorPoint = selected.AnchorPoint
	boundingBoxFrame.Position = selected.Position
	boundingBoxFrame.Size = selected.Size
	boundingBoxFrame.Rotation = selected.Rotation
	boundingBoxFrame.ZIndex = (tonumber(selected.ZIndex) or 1) + 50

	local hs = HANDLE_SIZE
	local positions = {
		TL = UDim2.new(0, -hs/2, 0, -hs/2),
		TR = UDim2.new(1, -hs/2, 0, -hs/2),
		BL = UDim2.new(0, -hs/2, 1, -hs/2),
		BR = UDim2.new(1, -hs/2, 1, -hs/2),
		T  = UDim2.new(0.5, -hs/2, 0, -hs/2),
		B  = UDim2.new(0.5, -hs/2, 1, -hs/2),
		L  = UDim2.new(0, -hs/2, 0.5, -hs/2),
		R  = UDim2.new(1, -hs/2, 0.5, -hs/2),
	}
	for name, handle in pairs(boundingHandles) do
		if handle and handle.Parent then
			handle.Position = positions[name]
			handle.ZIndex = boundingBoxFrame.ZIndex + 1
		end
	end
end

function createBoundingBox(inst)
	destroyBoundingBox()
	if not inst or not inst:IsA("GuiObject") then return end

	-- Fully transparent frame (outline only)
	boundingBoxFrame = new("Frame", {
		Name = "__BoundingBox",
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 10000,
		Active = true,
		Parent = inst.Parent or screenGui,
	})

	new("UIStroke", {
		Color = ORANGE,
		Thickness = 2,
		Transparency = 0.6, -- opacidade 0.40
		Parent = boundingBoxFrame,
	})

	local function makeHandle(name)
		local h = new("Frame", {
			Name = "__Handle_" .. name,
			Size = UDim2.fromOffset(HANDLE_SIZE, HANDLE_SIZE),
			BackgroundColor3 = ORANGE,
			BackgroundTransparency = 0.6, -- opacidade 0.40, sem contorno (sem UIStroke)
			BorderSizePixel = 0,
			ZIndex = 10001,
			Active = true,
			Parent = boundingBoxFrame,
		})
		new("UICorner", { CornerRadius = UDim.new(1, 0), Parent = h })
		boundingHandles[name] = h
		return h
	end

	local hTL = makeHandle("TL")
	local hTR = makeHandle("TR")
	local hBL = makeHandle("BL")
	local hBR = makeHandle("BR")
	local hT  = makeHandle("T")
	local hB  = makeHandle("B")
	local hL  = makeHandle("L")
	local hR  = makeHandle("R")

	updateBoundingBox()

	local function startDrag(input)
		if not selected or not selected:IsA("GuiObject") then return end
		isDraggingBB = true
		isResizingBB = false
		resizeMode = nil
		dragStartPos = input.Position
		startUDimPos = selected.Position
		startAbsPos = selected.AbsolutePosition
		startAbsSize = selected.AbsoluteSize
	end

	local function doDrag(input)
		if not isDraggingBB or not selected then return end
		local delta = input.Position - dragStartPos
		-- Proposed abs position from mouse delta
						local proposedAbsPos = Vector2.new(startAbsPos.X + delta.X, startAbsPos.Y + delta.Y)
		local snappedAbsPos = snapToGuides(selected, proposedAbsPos, startAbsSize)
		local finalDelta = Vector2.new(snappedAbsPos.X - startAbsPos.X, snappedAbsPos.Y - startAbsPos.Y)
		-- Direct delta on Position (works anywhere)
		selected.Position = UDim2.new(
			startUDimPos.X.Scale,
			startUDimPos.X.Offset + finalDelta.X,
			startUDimPos.Y.Scale,
			startUDimPos.Y.Offset + finalDelta.Y
		)
		updateBoundingBox()
	end

	local function startResize(input, mode)
		if not selected or not selected:IsA("GuiObject") then return end
		isResizingBB = true
		isDraggingBB = false
		resizeMode = mode
		dragStartPos = input.Position
		startAbsPos = selected.AbsolutePosition
		startAbsSize = selected.AbsoluteSize
		startUDimPos = selected.Position
		startUDimSize = selected.Size
	end

	local function doResize(input)
		if not isResizingBB or not selected or not resizeMode then return end
		local delta = input.Position - dragStartPos
		local ap = selected.AnchorPoint
		local newW = startAbsSize.X
		local newH = startAbsSize.Y

		local moveL = (resizeMode == "L" or resizeMode == "TL" or resizeMode == "BL")
		local moveR = (resizeMode == "R" or resizeMode == "TR" or resizeMode == "BR")
		local moveT = (resizeMode == "T" or resizeMode == "TL" or resizeMode == "TR")
		local moveB = (resizeMode == "B" or resizeMode == "BL" or resizeMode == "BR")

		if moveR then newW = math.max(16, startAbsSize.X + delta.X) end
		if moveL then newW = math.max(16, startAbsSize.X - delta.X) end
		if moveB then newH = math.max(16, startAbsSize.Y + delta.Y) end
		if moveT then newH = math.max(16, startAbsSize.Y - delta.Y) end

		-- Keep opposite edge fixed
		local ox, oy = 0, 0
		if moveL and not moveR then
			ox = (startAbsSize.X - newW) * (1 - ap.X)
		elseif moveR and not moveL then
			ox = (newW - startAbsSize.X) * ap.X
		end
		if moveT and not moveB then
			oy = (startAbsSize.Y - newH) * (1 - ap.Y)
		elseif moveB and not moveT then
			oy = (newH - startAbsSize.Y) * ap.Y
		end

		selected.Position = UDim2.new(
			startUDimPos.X.Scale, startUDimPos.X.Offset + ox,
			startUDimPos.Y.Scale, startUDimPos.Y.Offset + oy
		)
		selected.Size = UDim2.fromOffset(newW, newH)
		updateBoundingBox()
	end

	local function endInteraction()
		isDraggingBB = false
		isResizingBB = false
		resizeMode = nil
		clearAlignmentLines()
		refreshProperties()
	end

	-- Drag body (move)
	table.insert(bbConnections, boundingBoxFrame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			startDrag(input)
			local conn
			conn = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					endInteraction()
					conn:Disconnect()
				end
			end)
		end
	end))

	local function connectHandle(handle, mode)
		table.insert(bbConnections, handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				startResize(input, mode)
				local conn
				conn = input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then
						endInteraction()
						conn:Disconnect()
					end
				end)
			end
		end))
	end

	connectHandle(hTL, "TL")
	connectHandle(hTR, "TR")
	connectHandle(hBL, "BL")
	connectHandle(hBR, "BR")
	connectHandle(hT, "T")
	connectHandle(hB, "B")
	connectHandle(hL, "L")
	connectHandle(hR, "R")

	table.insert(bbConnections, UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			if isDraggingBB then
				doDrag(input)
			elseif isResizingBB then
				doResize(input)
			end
		end
	end))

	table.insert(bbConnections, RunService.RenderStepped:Connect(function()
		if selected and selected:IsA("GuiObject") and selected.Parent and boundingBoxFrame then
			if not isDraggingBB and not isResizingBB then
				updateBoundingBox()
			end
		end
	end))
end

local SAVES_ROOT_NAME = "__GuiBuilderSaves"
-- Pasta global no filesystem do executor (persiste entre jogos/places)
local DISK_SAVE_FOLDER = "GuiBuilderTool_Saves"
local canUseDisk = (typeof(writefile) == "function" and typeof(readfile) == "function"
	and typeof(isfile) == "function" and typeof(loadstring) == "function")

-- DataStore via servidor (persiste em qualquer jogo do mesmo lugar/experiência)
local saveRemote = nil
pcall(function()
	local rs = game:GetService("ReplicatedStorage")
	saveRemote = rs:FindFirstChild("GuiBuilderToolSaveRF")
end)

local function ensureDiskFolder()
	if canUseDisk and makefolder and isfolder then
		local ok, exists = pcall(isfolder, DISK_SAVE_FOLDER)
		if ok and not exists then
			pcall(makefolder, DISK_SAVE_FOLDER)
		end
	end
end

local function diskSavePath(username, saveName)
	local safeUser = username:gsub("[^%w_%-]", "_")
	local safeName = saveName:gsub("[^%w_%-]", "_")
	return DISK_SAVE_FOLDER .. "/" .. safeUser .. "__" .. safeName .. ".txt"
end

-- Serialize instances to Lua source (like Gui2Lua)
local function serializeTreeToLua(instances)
	local lines = {}
	local varCount = 0
	local function nextVar(name)
		varCount = varCount + 1
		local safe = name:gsub("[^%w_]", "")
		if safe == "" then safe = "Obj" end
		return safe .. varCount
	end
	table.insert(lines, "local __Root = Instance.new(\"Folder\")")
	local function exportInstance(inst, parentVar)
		if isOverlay(inst) then return end
		local className = inst.ClassName
		local props = EXPORTABLE_PROPS[className]
		local varName = nextVar(inst.Name)
		table.insert(lines, string.format("local %s = Instance.new(%q)", varName, className))
		if props then
			for _, propName in ipairs(props) do
				local ok, value = pcall(function() return inst[propName] end)
				if ok and value ~= nil then
					table.insert(lines, string.format("%s.%s = %s", varName, propName, luaValue(value)))
				end
			end
		else
			table.insert(lines, string.format("%s.Name = %q", varName, inst.Name))
		end
		table.insert(lines, string.format("%s.Parent = %s", varName, parentVar))
		for _, child in ipairs(inst:GetChildren()) do
			exportInstance(child, varName)
		end
	end
	for _, inst in ipairs(instances) do
		exportInstance(inst, "__Root")
	end
	table.insert(lines, "return __Root")
	return table.concat(lines, "\n")
end

local function getSavesRoot()
	local root = PlayerGui:FindFirstChild(SAVES_ROOT_NAME)
	if not root then
		root = Instance.new("Folder")
		root.Name = SAVES_ROOT_NAME
		root.Parent = PlayerGui
	end
	return root
end

local function getUserSavesFolder()
	local root = getSavesRoot()
	local uname = LocalPlayer.Name
	local folder = root:FindFirstChild(uname)
	if not folder then
		folder = Instance.new("Folder")
		folder.Name = uname
		folder.Parent = root

end

-- List saves: disk first, then session
local function listUserSaves()
	local names = {}
	local seen = {}

	local function addName(saveName)
		if saveName and saveName ~= "" and not seen[saveName] and not saveName:match("^__") then
			seen[saveName] = true
			table.insert(names, saveName)
		end
	end

	if canUseDisk and typeof(listfiles) == "function" then
		ensureDiskFolder()
		local ok, files = pcall(listfiles, DISK_SAVE_FOLDER)
		if ok and files then
			local prefix = LocalPlayer.Name:gsub("[^%w_%-]", "_") .. "__"
			for _, filePath in ipairs(files) do
				local fileName = tostring(filePath):match("([^/\\]+)$") or tostring(filePath)
				if fileName:sub(1, #prefix) == prefix then
					addName(fileName:sub(#prefix + 1):gsub("%.txt$", ""))
				end
			end
		end
	end

	if saveRemote then
		local ok, remoteNames = pcall(function()
			return saveRemote:InvokeServer("List")
		end)
		if ok and type(remoteNames) == "table" then
			for _, n in ipairs(remoteNames) do
				addName(n)
			end
		end
	end

	local folder = getUserSavesFolder()
	for _, child in ipairs(folder:GetChildren()) do
		if child:IsA("Folder") then
			addName(child.Name)
		end
	end

	table.sort(names)
	return names
end

local function captureProjectSnapshot()
	local clones = {}
	if not projectGui then return clones end
	for _, child in ipairs(projectGui:GetChildren()) do
		if not isOverlay(child) then
			table.insert(clones, child:Clone())
		end
	end
	return clones
end

local function writeSave(saveName, clones)
	-- session copy (fast)
	local folder = getUserSavesFolder()
	local existing = folder:FindFirstChild(saveName)
	if existing then existing:Destroy() end
	local saveFolder = Instance.new("Folder")
	saveFolder.Name = saveName
	saveFolder.Parent = folder
	for _, c in ipairs(clones) do
		c.Parent = saveFolder
	end

	-- Persistência real: disco (executor) + DataStore (servidor)
	local okSrc, srcCode = pcall(serializeTreeToLua, clones)
	if okSrc and srcCode then
		if canUseDisk then
			ensureDiskFolder()
			pcall(writefile, diskSavePath(LocalPlayer.Name, saveName), srcCode)
		end
		if saveRemote then
			task.spawn(function()
				pcall(function()
					saveRemote:InvokeServer("Save", saveName, srcCode)
				end)
			end)
		end
	end

	return saveFolder
end

local function loadSave(saveName)
	local folder = getUserSavesFolder()
	local saveFolder = folder:FindFirstChild(saveName)
	local sourceInstances = nil
	local tempRoot = nil

	if saveFolder then
		sourceInstances = saveFolder:GetChildren()
	else
		local srcCode = nil
		if canUseDisk then
			local path = diskSavePath(LocalPlayer.Name, saveName)
			local ok, exists = pcall(isfile, path)
			if ok and exists then
				local ok2, content = pcall(readfile, path)
				if ok2 then srcCode = content end
			end
		end
		if (not srcCode or srcCode == "") and saveRemote then
			local okR, content = pcall(function()
				return saveRemote:InvokeServer("Load", saveName)
			end)
			if okR and type(content) == "string" and content ~= "" then
				srcCode = content
			end
		end
		if srcCode and srcCode ~= "" then
			local chunk = loadstring(srcCode)
			if chunk then
				local ok3, root = pcall(chunk)
				if ok3 and root then
					tempRoot = root
					sourceInstances = root:GetChildren()
				end
			end
		end
	end

	if not sourceInstances then return false end

	selected = nil
	destroyBoundingBox()
	clearAlignmentLines()
	for _, child in ipairs(projectGui:GetChildren()) do
		if not isOverlay(child) then
			child:Destroy()
		end
	end
	for _, inst in ipairs(sourceInstances) do
		inst:Clone().Parent = projectGui
	end
	if tempRoot then
		tempRoot:Destroy()
	end
	refreshExplorer()
	selectInstance(projectGui)
	return true
end

local function deleteSave(saveName)
	local removedSomething = false
	local folder = getUserSavesFolder()
	local saveFolder = folder:FindFirstChild(saveName)
	if saveFolder then
		saveFolder:Destroy()
		removedSomething = true
	end
	if canUseDisk then
		local path = diskSavePath(LocalPlayer.Name, saveName)
		local ok, exists = pcall(isfile, path)
		if ok and exists and typeof(delfile) == "function" then
			pcall(delfile, path)
			removedSomething = true
		end
	end
	if saveRemote then
		local ok = pcall(function()
			return saveRemote:InvokeServer("Delete", saveName)
		end)
		if ok then removedSomething = true end
	end
	return removedSomething
end

-- AUTOSAVE: keeps the active project (not the named saves list) on disk,
-- so it's still there after leaving/rejoining the game.
local function autosavePath()
	local safeUser = LocalPlayer.Name:gsub("[^%w_%-]", "_")
	return DISK_SAVE_FOLDER .. "/__autosave_" .. safeUser .. ".txt"
end

local function doAutosave()
	if not canUseDisk then return end
	ensureDiskFolder()
	local clones = captureProjectSnapshot()
	local ok, src = pcall(serializeTreeToLua, clones)
	for _, c in ipairs(clones) do
		c:Destroy()
	end
	if ok then
		pcall(writefile, autosavePath(), src)
	end
end

function loadAutosaveIfExists()
	if not canUseDisk then return false end
	local path = autosavePath()
	local ok, exists = pcall(isfile, path)
	if not (ok and exists) then return false end
	local ok2, src = pcall(readfile, path)
	if not (ok2 and src) or src == "" then return false end
	local chunk = loadstring(src)
	if not chunk then return false end
	local ok3, root = pcall(chunk)
	if not (ok3 and root) then return false end
	for _, child in ipairs(projectGui:GetChildren()) do
		if not isOverlay(child) then
			child:Destroy()
		end
	end
	for _, inst in ipairs(root:GetChildren()) do
		inst:Clone().Parent = projectGui
	end
	root:Destroy()
	return true
end

-- Autosaves every few seconds while the tool is open (simpler and safer
-- than trying to hook every single edit path).
task.spawn(function()
	while true do
		task.wait(6)
		pcall(doAutosave)
	end
end)

	for _, child in ipairs(savesScroll:GetChildren()) do

	end

	local names = listUserSaves()

				Font = Enum.Font.Gotham,

				TextColor3 = THEME.Text,
				TextSize = 13,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,

			local editBtn = makeSaveActionBtn(row, ICONS.EditSave, Color3.fromRGB(200, 180, 40), -116)

				if loadSave(saveName) then

					saveLoadWindow.Visible = false
				end
			end)

			local loadBtn = makeSaveActionBtn(row, ICONS.LoadSave, Color3.fromRGB(50, 180, 70), -86)
			loadBtn.MouseButton1Click:Connect(function()
				if loadSave(saveName) then
					saveLoadStatus.Text = "Loaded: " .. saveName

-- Mobile clipboard has a size limit, so split code into chunks
local MOBILE_CODE_CHUNK_LIMIT = 6000
local codeChunks = {""}
local codeChunkIndex = 1

local function splitCodeIntoChunks(code, limit)
	if #code <= limit then
		return {code}
	end
	local chunks = {}
	local current = {}
	local currentLen = 0
	for line in (code .. "\n"):gmatch("([^\n]*)\n") do
		local lineLen = #line + 1
		if currentLen + lineLen > limit and #current > 0 then
			table.insert(chunks, table.concat(current, "\n"))
			current = {}
			currentLen = 0
		end
		table.insert(current, line)
		currentLen = currentLen + lineLen
	end
	if #current > 0 then
		table.insert(chunks, table.concat(current, "\n"))
	end
	return chunks
end

local function showCodeChunk()
	codeOutputBox.Text = codeChunks[codeChunkIndex] or ""
	codePageLabel.Text = string.format("Parte %d/%d", codeChunkIndex, #codeChunks)
end

local function updateCodeOutputView()
	local isMobileNow = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
	if isMobileNow and #codeChunks > 1 then
		codePaginationRow.Visible = true
		codeOutputScroll.Position = codeScrollPagedPos
		codeOutputScroll.Size = codeScrollPagedSize
	else
		codePaginationRow.Visible = false
		codeOutputScroll.Position = codeScrollNormalPos
		codeOutputScroll.Size = codeScrollNormalSize
	end
	showCodeChunk()
end

local gradientEditorTarget = nil

local function sortGradientKeypoints()
	table.sort(gradientKeypoints, function(a, b) return a.Time < b.Time end)
	local n = #gradientKeypoints
	if n > 0 then
		gradientKeypoints[1].Time = 0
		gradientKeypoints[n].Time = 1
	end
	for i = 2, n - 1 do
		if gradientKeypoints[i].Time <= gradientKeypoints[i - 1].Time then
			gradientKeypoints[i].Time = gradientKeypoints[i - 1].Time + 0.0001
		end
	end
	if n >= 2 and gradientKeypoints[n - 1].Time >= gradientKeypoints[n].Time then
		gradientKeypoints[n - 1].Time = gradientKeypoints[n].Time - 0.0001
	end
end

local function buildColorSequence()
	local seq = {}
	for _, kp in ipairs(gradientKeypoints) do
		table.insert(seq, ColorSequenceKeypoint.new(kp.Time, kp.Color))
	end
	return seq
end

local function applyGradientToInstance()
	if not gradientEditorTarget then return end
	local seq = buildColorSequence()
	local ok = pcall(function()
		gradientEditorTarget.Color = ColorSequence.new(seq)
		gradientBarPreview.Color = ColorSequence.new(seq)
	end)
	if not ok and #gradientKeypoints >= 2 then
		local fallback = ColorSequence.new(gradientKeypoints[1].Color, gradientKeypoints[#gradientKeypoints].Color)
		gradientEditorTarget.Color = fallback
		gradientBarPreview.Color = fallback
	end
end

local function refreshGradientMarkers()
	for _, marker in ipairs(gradientMarkers) do
		marker:Destroy()
	end
	gradientMarkers = {}

	for _, kp in ipairs(gradientKeypoints) do
		local marker = new("ImageButton", {
			Name = "Marker",
			Size = UDim2.new(0, 16, 0, 16),
			AnchorPoint = Vector2.new(0.5, 0),
			Position = UDim2.new(kp.Time, 0, 0, 0),
			BackgroundTransparency = 1,
			Image = GRADIENT_MARKER_IMAGE,
			ImageColor3 = (kp == selectedKeypoint) and Color3.fromRGB(255, 221, 84) or Color3.fromRGB(190, 190, 190),
			ZIndex = 20,
			Parent = gradientMarkerHolder,
		})

		marker.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				selectedKeypoint = kp
				draggingKeypoint = kp
				gradientTimeBox.Text = tostring(round(kp.Time, 3))
				gradientColorSwatch.BackgroundColor3 = kp.Color
				refreshGradientMarkers()
			end
		end)

		table.insert(gradientMarkers, marker)
	end
end

local function exportToLua(options)
	options = options or {}
	local lines = {}
	local varCount = 0

	local function nextVar(name)
		varCount = varCount + 1
		local safe = name:gsub("[^%w_]", "")
		if safe == "" then safe = "Obj" end
		return safe .. varCount
	end

	local rootClassName = projectGui.ClassName
	table.insert(lines, string.format("local ProjectGui = Instance.new(%q)", rootClassName))
	table.insert(lines, string.format("ProjectGui.Name = %q", projectGui.Name))
	if rootClassName == "ScreenGui" then
		table.insert(lines, "ProjectGui.ResetOnSpawn = false")
	end
	table.insert(lines, "ProjectGui.Parent = game:GetService(\"Players\").LocalPlayer:WaitForChild(\"PlayerGui\")")
	table.insert(lines, "")

	local function exportInstance(inst, parentVar)
		if isOverlay(inst) then return end
		local className = inst.ClassName
		local props = EXPORTABLE_PROPS[className]
		local varName = nextVar(inst.Name)

		if options.comments then
			table.insert(lines, string.format("-- %s: %q", className, inst.Name))
		end
		table.insert(lines, string.format("local %s = Instance.new(%q)", varName, className))

		if props then
			for _, propName in ipairs(props) do
				local ok, value = pcall(function() return inst[propName] end)
				if ok and value ~= nil then
					if propName == "Size" and options.autoSizeScale and inst:IsA("GuiObject") then
						local parentObj = inst.Parent
						local parentSize = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsoluteSize or inst.AbsoluteSize
						local sx = parentSize.X > 0 and (inst.AbsoluteSize.X / parentSize.X) or 0
						local sy = parentSize.Y > 0 and (inst.AbsoluteSize.Y / parentSize.Y) or 0
						table.insert(lines, string.format("%s.Size = UDim2.new(%s, 0, %s, 0)", varName, round(sx, 4), round(sy, 4)))
					elseif propName == "Position" and options.autoPositionScale and inst:IsA("GuiObject") then
						local parentObj = inst.Parent
						local parentSize = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsoluteSize or inst.AbsoluteSize
						local parentPos = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsolutePosition or Vector2.new(0, 0)
						local relX = inst.AbsolutePosition.X - parentPos.X
						local relY = inst.AbsolutePosition.Y - parentPos.Y
						local px = parentSize.X > 0 and (relX / parentSize.X) or 0
						local py = parentSize.Y > 0 and (relY / parentSize.Y) or 0
						table.insert(lines, string.format("%s.Position = UDim2.new(%s, 0, %s, 0)", varName, round(px, 4), round(py, 4)))
					else
						table.insert(lines, string.format("%s.%s = %s", varName, propName, luaValue(value)))
					end
				end
			end
		else
			table.insert(lines, string.format("%s.Name = %q", varName, inst.Name))
		end

		table.insert(lines, string.format("%s.Parent = %s", varName, parentVar))

		if options.autoAddConstraint and inst:IsA("GuiObject") and inst.AbsoluteSize.Y > 0 then
			local ratio = inst.AbsoluteSize.X / inst.AbsoluteSize.Y
			local constraintVar = varName .. "Ratio"
			table.insert(lines, string.format("local %s = Instance.new(\"UIAspectRatioConstraint\")", constraintVar))
			table.insert(lines, string.format("%s.AspectRatio = %s", constraintVar, round(ratio, 3)))
			table.insert(lines, string.format("%s.Parent = %s", constraintVar, varName))
		end
		table.insert(lines, "")
		for _, child in ipairs(inst:GetChildren()) do
			exportInstance(child, varName)
		end
	end

	for _, child in ipairs(projectGui:GetChildren()) do
		exportInstance(child, "ProjectGui")
	end
	return table.concat(lines, "\n")
end

local function makeHorizontalDraggable(frame)
	local dragging = false
	local dragInput, dragStart, startPos

	local function update(input)
		if not dragging then return end
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset)
	end

	frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			update(input)
		end
	end)
end

-- Maximize/minimize tween
local optionsMenuOpen = false
local optionsMenuOriginalSize = UDim2.new(0, 180, 0, OPTIONS_MENU_HEIGHT)
local optionsMenuMinimizedSize = UDim2.new(0, 180, 0, 0)
optionsMenu.Size = optionsMenuMinimizedSize

-- Enquanto o menu estiver aberto, a posição é recalculada every frame,
-- então ele nunca "volta pra cima" nem fica preso atrás do botão,
-- não importa o que tenha acontecido antes (drag, resize, etc).
local optionsMenuSyncConn = nil
local function setOptionsMenuSyncing(syncing)
	if optionsMenuSyncConn then
		optionsMenuSyncConn:Disconnect()
		optionsMenuSyncConn = nil
	end
	if syncing then
		repositionOptionsMenu()
		optionsMenuSyncConn = RunService.RenderStepped:Connect(repositionOptionsMenu)
	end
end

optionsButton.MouseButton1Click:Connect(function()
	optionsMenuOpen = not optionsMenuOpen
	setOptionsMenuSyncing(optionsMenuOpen)

	local targetSize = optionsMenuOpen and optionsMenuOriginalSize or optionsMenuMinimizedSize
	local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
	TweenService:Create(optionsMenu, tweenInfo, { Size = targetSize }):Play()
end)
end -- do (OPTIONS MENU)

function clearHighlight()
	if selected then
		local overlay = selected:FindFirstChild("__SelectionOverlay")
		if overlay then overlay:Destroy() end
	end
end

function highlight(inst) end

function selectInstance(inst)
	if selected == inst then return end
	clearHighlight()
	selected = inst
	destroyBoundingBox()
	if inst and inst:IsA("GuiObject") and inst ~= projectGui then
		createBoundingBox(inst)
	end
	refreshProperties()

	for _, row in pairs(explorerRows) do
		if row.Instance == selected then
			row.Button.BackgroundColor3 = THEME.RowSelected
		else
			row.Button.BackgroundColor3 = THEME.Panel
		end
	end
end

function buildPropertyRow(schemaEntry, order)
	local ok, value = pcall(function() return selected[schemaEntry.Key] end)
	if not ok and schemaEntry.Type ~= "source" then return nil end

	-- Source: LocalScript/ModuleScript only
	if schemaEntry.Type == "source" then
		if not (selected:IsA("LocalScript") or selected:IsA("ModuleScript")) then
			return nil
		end
	end

	local row = new("Frame", {
		Size = UDim2.new(1, 0, 0, 22),
		BackgroundTransparency = 1,
		LayoutOrder = order,
	})
	new("TextLabel", {
		Size = UDim2.new(0.46, 0, 1, 0),
		BackgroundTransparency = 1,
		Font = Enum.Font.Gotham,
		Text = schemaEntry.Label,
		TextColor3 = THEME.SubText,
		TextSize = 11,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = row,
	})

	local t = schemaEntry.Type

	if t == "source" then
		local btn = new("TextButton", {
			Size = UDim2.new(0.54, 0, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.Accent,
			Font = Enum.Font.GothamBold,
			Text = "Use / Edit",
			TextColor3 = Color3.fromRGB(255, 255, 255),
			TextSize = 11,
			Parent = row,
		})
		new("UICorner", { CornerRadius = UDim.new(0, 4), Parent = btn })
		btn.MouseButton1Click:Connect(function()
			openSourceEditor(selected)
		end)

	elseif t == "readonly" then
		local text
		local vt = typeof(value)
		if vt == "Vector2" then
			text = string.format("%d, %d", round(value.X, 0), round(value.Y, 0))
		elseif vt == "number" then
			text = tostring(round(value, 2))
		else
			text = tostring(value)
		end
		new("TextLabel", {
			Size = UDim2.new(0.54, 0, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundTransparency = 1,
			Font = Enum.Font.Gotham,
			Text = text,
			TextColor3 = THEME.SubText,
			TextSize = 11,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextTruncate = Enum.TextTruncate.AtEnd,
			Parent = row,
		})

	elseif t == "bool" then
		local box = new("ImageButton", {
			Size = UDim2.new(0, 20, 0, 20),
			Position = UDim2.new(0.46, 0, 0, 1),
			BackgroundColor3 = THEME.PanelAlt,
			Image = value and ICONS.ToggleCheck or "",
			ScaleType = Enum.ScaleType.Fit,
			Parent = row,
		})
		new("UICorner", { CornerRadius = UDim.new(0, 4), Parent = box })
		box.MouseButton1Click:Connect(function()
			local newValue = not selected[schemaEntry.Key]
			selected[schemaEntry.Key] = newValue
			box.Image = newValue and ICONS.ToggleCheck or ""
		end)

	elseif t == "string" then
		local box = new("TextBox", {
			Size = UDim2.new(0.54, 0, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			ClearTextOnFocus = false,
			Font = Enum.Font.Gotham,
			Text = tostring(value),
			TextColor3 = THEME.Text,
			TextSize = 11,
			Parent = row,
		})
		box.FocusLost:Connect(function()
			local text = box.Text
			-- Image/Video: raw asset id is enough
						if (schemaEntry.Key == "Image" or schemaEntry.Key == "Video") and text ~= "" then
				if not (text:match("^rbxassetid://") or text:match("^https?://") or text:match("^rbxthumb://")) then
					local digits = text:match("^%s*(%d+)%s*$") or text:match("(%d+)")
					if digits then
						text = "rbxassetid://" .. digits
					end
				end
				box.Text = text
			end
			selected[schemaEntry.Key] = text
		end)

	elseif t == "number" then
		local box = new("TextBox", {
			Size = UDim2.new(0.54, 0, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			ClearTextOnFocus = false,
			Font = Enum.Font.Gotham,
			Text = tostring(round(value, 3)),
			TextColor3 = THEME.Text,
			TextSize = 11,
			Parent = row,
		})
		box.FocusLost:Connect(function()
			local n = tonumber(box.Text)
			if n then selected[schemaEntry.Key] = n end
		end)

	elseif t == "color" then
		if typeof(value) == "ColorSequence" then
			local gradientButton = new("TextButton", {
				Size = UDim2.new(0.54, 0, 0, 20),
				Position = UDim2.new(0.46, 0, 0, 0),
				BackgroundColor3 = THEME.PanelAlt,
				AutoButtonColor = false,
				Font = Enum.Font.Gotham,
				Text = "Gradient",
				TextColor3 = THEME.Text,
				TextSize = 11,
				Parent = row,
			})
			gradientButton.MouseEnter:Connect(function() gradientButton.BackgroundColor3 = THEME.ButtonHover end)
			gradientButton.MouseLeave:Connect(function() gradientButton.BackgroundColor3 = THEME.PanelAlt end)
			gradientButton.MouseButton1Click:Connect(function()
				openGradientEditor(selected)
			end)
		else
			local swatch = new("TextButton", {
				Size = UDim2.new(0.54, 0, 0, 18),
				Position = UDim2.new(0.46, 0, 0, 1),
				BackgroundColor3 = value,
				AutoButtonColor = false,
				Text = "",
				Parent = row,
			})
			new("UIStroke", { Color = THEME.Divider, Thickness = 1, Parent = swatch })
			swatch.MouseButton1Click:Connect(function()
				openColorPicker(selected[schemaEntry.Key], function(newColor)
					selected[schemaEntry.Key] = newColor
					swatch.BackgroundColor3 = newColor
				end)
			end)
		end

	elseif t == "vector2" then
		local boxes = {}
		local values = {value.X, value.Y}
		for i = 1, 2 do
			boxes[i] = new("TextBox", {
				Size = UDim2.new(0.27, 0, 1, 0),
				Position = UDim2.new(0.46 + (i - 1) * 0.27, 0, 0, 0),
				BackgroundColor3 = THEME.PanelAlt,
				Font = Enum.Font.Gotham,
				Text = tostring(round(values[i], 3)),
				TextColor3 = THEME.Text,
				TextSize = 11,
				Parent = row,
			})
		end
		local function commit()
			local x = tonumber(boxes[1].Text) or selected[schemaEntry.Key].X
			local y = tonumber(boxes[2].Text) or selected[schemaEntry.Key].Y
			selected[schemaEntry.Key] = Vector2.new(x, y)
		end
		boxes[1].FocusLost:Connect(commit)
		boxes[2].FocusLost:Connect(commit)

	elseif t == "udim" then
		local boxes = {}
		local values = {value.Scale, value.Offset}
		for i = 1, 2 do
			boxes[i] = new("TextBox", {
				Size = UDim2.new(0.27, 0, 1, 0),
				Position = UDim2.new(0.46 + (i - 1) * 0.27, 0, 0, 0),
				BackgroundColor3 = THEME.PanelAlt,
				Font = Enum.Font.Gotham,
				Text = tostring(round(values[i], 3)),
				TextColor3 = THEME.Text,
				TextSize = 11,
				Parent = row,
			})
		end
		local function commit()
			local s = tonumber(boxes[1].Text) or selected[schemaEntry.Key].Scale
			local o = tonumber(boxes[2].Text) or selected[schemaEntry.Key].Offset
			selected[schemaEntry.Key] = UDim.new(s, o)
		end
		boxes[1].FocusLost:Connect(commit)
		boxes[2].FocusLost:Connect(commit)

	elseif t == "udim2" then
		row.Size = UDim2.new(1, 0, 0, 42)
		local boxes = {}
		local values = {value.X.Scale, value.X.Offset, value.Y.Scale, value.Y.Offset}
		for i = 1, 4 do
			boxes[i] = new("TextBox", {
				Size = UDim2.new(0.24, -2, 0, 18),
				Position = UDim2.new((i - 1) * 0.25, 0, 0, 22),
				BackgroundColor3 = THEME.PanelAlt,
				Font = Enum.Font.Gotham,
				Text = tostring(round(values[i], 3)),
				TextColor3 = THEME.Text,
				TextSize = 11,
				Parent = row,
			})
		end
		local function commit()
			local xs = tonumber(boxes[1].Text) or selected[schemaEntry.Key].X.Scale
			local xo = tonumber(boxes[2].Text) or selected[schemaEntry.Key].X.Offset
			local ys = tonumber(boxes[3].Text) or selected[schemaEntry.Key].Y.Scale
			local yo = tonumber(boxes[4].Text) or selected[schemaEntry.Key].Y.Offset
			selected[schemaEntry.Key] = UDim2.new(xs, xo, ys, yo)
			if selected:IsA("GuiObject") then
				updateBoundingBox()
			end
		end
		for _, box in ipairs(boxes) do
			box.FocusLost:Connect(commit)
		end

	elseif t == "enum" then
		local box = new("TextButton", {
			Size = UDim2.new(0.54, 0, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			Font = Enum.Font.Gotham,
			Text = value.Name,
			TextColor3 = THEME.Text,
			TextSize = 11,
			Parent = row,
		})
		box.MouseButton1Click:Connect(function()
			local items = schemaEntry.Enum:GetEnumItems()
			local currentValue = selected[schemaEntry.Key]
			local currentIndex = 1
			for i, item in ipairs(items) do
				if item == currentValue then
					currentIndex = i
					break
				end
			end
			local nextItem = items[(currentIndex % #items) + 1]
			selected[schemaEntry.Key] = nextItem
			box.Text = nextItem.Name
		end)
	end

	return row
end

function refreshProperties()
	for _, child in ipairs(propertiesList:GetChildren()) do
		if child:IsA("GuiObject") then
			child:Destroy()
		end
	end
	if not selected then return end

	propertiesTitle.Text = string.format('properties - %s "%s"',
		CLASS_LABELS_PT[selected.ClassName] or selected.ClassName, selected.Name)

	local searchText = propertiesSearch.Text:lower()
	local order = 0
	for _, entry in ipairs(PROPERTY_SCHEMA) do
		if searchText == "" or entry.Label:lower():find(searchText, 1, true) then
			local row = buildPropertyRow(entry, order + 1)
			if row then
				order = order + 1
				row.Parent = propertiesList
			end
		end
	end

	-- Corner control (round/rectangular), ScrollingFrame only
		if selected.ClassName == "ScrollingFrame" and (searchText == "" or ("canto"):find(searchText, 1, true)) then
		order = order + 1
		local cornerRow = new("Frame", {
			Size = UDim2.new(1, 0, 0, 20),
			BackgroundTransparency = 1,
			LayoutOrder = order,
			Parent = propertiesList,
		})
		new("TextLabel", {
			Size = UDim2.new(0.46, 0, 1, 0),
			BackgroundTransparency = 1,
			Font = Enum.Font.Gotham,
			Text = "Corner",
			TextColor3 = THEME.SubText,
			TextSize = 11,
			TextXAlignment = Enum.TextXAlignment.Left,
			Parent = cornerRow,
		})

		local existingCorner = selected:FindFirstChildOfClass("UICorner")
		local cornerRadiusBox
		local cornerToggle = new("TextButton", {
			Size = UDim2.new(0.28, -2, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			Font = Enum.Font.Gotham,
			Text = existingCorner and "Arredondado" or "Retangular",
			TextColor3 = THEME.Text,
			TextSize = 10,
			Parent = cornerRow,
		})
		cornerRadiusBox = new("TextBox", {
			Size = UDim2.new(0.26, 0, 1, 0),
			Position = UDim2.new(0.74, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			ClearTextOnFocus = false,
			Font = Enum.Font.Gotham,
			Text = existingCorner and tostring(existingCorner.CornerRadius.Offset) or "0",
			TextColor3 = THEME.Text,
			TextSize = 10,
			Visible = existingCorner ~= nil,
			Parent = cornerRow,
		})

		cornerToggle.MouseButton1Click:Connect(function()
			local current = selected:FindFirstChildOfClass("UICorner")
			if current then
				current:Destroy()
				cornerToggle.Text = "Rectangular"
				cornerRadiusBox.Visible = false
			else
				new("UICorner", {
					CornerRadius = UDim.new(0, 8),
					Parent = selected,
				})
				cornerToggle.Text = "Rounded"
				cornerRadiusBox.Text = "8"
				cornerRadiusBox.Visible = true
			end
			refreshExplorer()
		end)

		cornerRadiusBox.FocusLost:Connect(function()
			local n = tonumber(cornerRadiusBox.Text)
			local current = selected:FindFirstChildOfClass("UICorner")
			if n and current then
				current.CornerRadius = UDim.new(0, n)
			end
		end)
	end
end

function clearExplorer()
	for _, row in pairs(explorerRows) do
		if row.Row then row.Row:Destroy() end
	end
	explorerRows = {}
end

local TREE_LINE_COLOR = Color3.fromRGB(255, 140, 0)
local TREE_STEP_PX = 14

function buildExplorer(inst, depth, order, ancestorIsLast)
	if isOverlay(inst) then return order end
	ancestorIsLast = ancestorIsLast or {}

	order = order + 1
	local row = new("Frame", {
		Name = "Row",
		Size = UDim2.new(1, 0, 0, 24),
		BackgroundColor3 = THEME.Panel,
		LayoutOrder = order,
		Parent = explorerList,
	})

	local indentPx = depth * TREE_STEP_PX

	-- Thin tree connector lines (parent -> child)
			if depth > 0 then
		for level = 1, depth - 1 do
			if not ancestorIsLast[level] then
				local colX = (level - 1) * TREE_STEP_PX + 7
				new("Frame", {
					Name = "TreeTrunk",
					Size = UDim2.new(0, 1, 1, 0),
					Position = UDim2.new(0, colX, 0, 0),
					BackgroundColor3 = TREE_LINE_COLOR,
					BorderSizePixel = 0,
					ZIndex = 2,
					Parent = row,
				})
			end
		end

		local myColX = (depth - 1) * TREE_STEP_PX + 7
		local isLastHere = ancestorIsLast[depth]

		-- top half (always)
		new("Frame", {
			Name = "TreeBranchTop",
			Size = UDim2.new(0, 1, 0, 12),
			Position = UDim2.new(0, myColX, 0, 0),
			BackgroundColor3 = TREE_LINE_COLOR,
			BorderSizePixel = 0,
			ZIndex = 2,
			Parent = row,
		})
		-- bottom half (only if more siblings follow)
				if not isLastHere then
			new("Frame", {
				Name = "TreeBranchBottom",
				Size = UDim2.new(0, 1, 0, 12),
				Position = UDim2.new(0, myColX, 0, 12),
				BackgroundColor3 = TREE_LINE_COLOR,
				BorderSizePixel = 0,
				ZIndex = 2,
				Parent = row,
			})
		end
		-- horizontal branch to the label
		new("Frame", {
			Name = "TreeBranchH",
			Size = UDim2.new(0, indentPx - myColX, 0, 1),
			Position = UDim2.new(0, myColX, 0, 12),
			BackgroundColor3 = TREE_LINE_COLOR,
			BorderSizePixel = 0,
			ZIndex = 2,
			Parent = row,
		})
	end

	local selectButton = new("TextButton", {
		Size = UDim2.new(1, -28 - indentPx, 1, 0),
		Position = UDim2.new(0, indentPx, 0, 0),
		BackgroundColor3 = (inst == selected) and THEME.RowSelected or THEME.Panel,
		AutoButtonColor = false,
		Font = Enum.Font.Gotham,
		Text = (CLASS_LABELS_PT[inst.ClassName] or inst.ClassName) .. ' "' .. inst.Name .. '"',
		TextColor3 = THEME.Text,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = row,
	})
	selectButton.MouseButton1Click:Connect(function()
		selectInstance(inst)
	end)

	local plusButton = new("ImageButton", {
		Size = UDim2.new(0, 20, 0, 20),
		Position = UDim2.new(1, -24, 0, 2),
		BackgroundColor3 = THEME.Button,
		Image = ICONS.Plus,
		ScaleType = Enum.ScaleType.Fit,
		ZIndex = 2,
		Parent = row,
	})
	new("UICorner", { CornerRadius = UDim.new(0, 4), Parent = plusButton })
	plusButton.MouseButton1Click:Connect(function()
		openAddElementPopup(inst, row)
	end)

	explorerRows[#explorerRows + 1] = { Row = row, Button = selectButton, Instance = inst }

	local visibleChildren = {}
	for _, child in ipairs(inst:GetChildren()) do
		if not isOverlay(child) then
			table.insert(visibleChildren, child)
		end
	end

	for i, child in ipairs(visibleChildren) do
		local childAncestorIsLast = {}
		for k, v in pairs(ancestorIsLast) do
			childAncestorIsLast[k] = v
		end
		childAncestorIsLast[depth + 1] = (i == #visibleChildren)
		order = buildExplorer(child, depth + 1, order, childAncestorIsLast)
	end
	return order
end

function refreshExplorer()
	clearExplorer()
	buildExplorer(projectGui, 0, 0)
end

function addElement(className, targetInstance)
	elementCounter = elementCounter + 1
	local defaults = CLASS_DEFAULTS[className] or {}
	local props = {}
	for k, v in pairs(defaults) do
		props[k] = v
	end
	props.Name = className .. elementCounter

	local parent = targetInstance or projectGui
	local inst = new(className, props)
	inst.Parent = parent

	if (className == "LocalScript" or className == "ModuleScript") and defaults.Source then
		inst.Source = defaults.Source
	end

	-- ScrollingFrame starts rounded (can go rectangular)
		if className == "ScrollingFrame" then
		new("UICorner", {
			CornerRadius = UDim.new(0, 8),
			Parent = inst,
		})
	end

	refreshExplorer()
	selectInstance(inst)
end

function applyTheme(themeName)
	if not THEMES[themeName] then return end
	currentThemeName = themeName
	for k, v in pairs(THEMES[themeName]) do
		THEME[k] = v
	end

	explorerWindow.BackgroundColor3 = THEME.Background
	propertiesWindow.BackgroundColor3 = THEME.Background
	settingsWindow.BackgroundColor3 = THEME.Background
	saveLoadWindow.BackgroundColor3 = THEME.Background
	gui2luaWindow.BackgroundColor3 = THEME.Background
	addElementPopup.BackgroundColor3 = THEME.Background
	codeOutputWindow.BackgroundColor3 = THEME.Panel
	-- optionsMenu color is fixed, not theme-based
	sourceEditorWindow.BackgroundColor3 = THEME.Background

	explorerTop.BackgroundColor3 = THEME.Panel
	propertiesTop.BackgroundColor3 = THEME.Panel
	settingsTop.BackgroundColor3 = THEME.Panel
	saveLoadTop.BackgroundColor3 = THEME.Panel
	gui2luaTop.BackgroundColor3 = THEME.Panel
	addElementTop.BackgroundColor3 = THEME.Panel
	codeOutputTop.BackgroundColor3 = THEME.PanelAlt
	sourceEditorTop.BackgroundColor3 = THEME.Panel
	actionBar.BackgroundColor3 = THEME.Panel

	for _, t in ipairs({themeToggleBlack, themeToggleWhite}) do
		local active = (t.themeName == currentThemeName)
		t.track.BackgroundColor3 = active and THEME.Accent or THEME.Button
		t.knob.Position = active and UDim2.new(1, -22, 0.5, -10) or UDim2.new(0, 2, 0.5, -10)
	end

	refreshExplorer()
	refreshProperties()
end

local coreGuiList = { "none" }

local function refreshGuiLists()
	coreGuiList = { "none" }
	playerGuiList = { "none" }
	-- CoreGui (may fail without permission)
	pcall(function()
		local CoreGui = game:GetService("CoreGui")
		for _, child in ipairs(CoreGui:GetChildren()) do
			if child:IsA("ScreenGui") or child:IsA("LayerCollector") or child:IsA("Folder") then
				table.insert(coreGuiList, child.Name)
			end
		end
	end)
	for _, child in ipairs(PlayerGui:GetChildren()) do
		if child:IsA("ScreenGui") and child.Name ~= "GuiBuilderTool" then
			table.insert(playerGuiList, child.Name)
		end
	end
	coreGuiDrop.Text = selectedCoreGuiName
	playerGuiDrop.Text = selectedPlayerGuiName
end

local function cycleDropdown(btn, list, current)
	local idx = 1
	for i, name in ipairs(list) do
		if name == current then idx = i break end
	end
	idx = (idx % #list) + 1
	btn.Text = list[idx]
	return list[idx]
end

	refreshGuiLists()
	selectedCoreGuiName = cycleDropdown(coreGuiDrop, coreGuiList, selectedCoreGuiName)
end)

playerGuiDrop.MouseButton1Click:Connect(function()
	refreshGuiLists()
	selectedPlayerGuiName = cycleDropdown(playerGuiDrop, playerGuiList, selectedPlayerGuiName)
end)

	if not gui then return end

	projectGui = gui
	selected = nil
	destroyBoundingBox()
	clearAlignmentLines()
	refreshExplorer()
	selectInstance(projectGui)
	explorerWindow.Visible = true
	propertiesWindow.Visible = true
end
