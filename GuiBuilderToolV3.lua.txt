local MAIN_WIDTH = 340
local MAIN_HEIGHT = 262
local HAZARD_HEIGHT = 14
local PERIOD = 28

local YELLOW = Color3.fromRGB(255, 205, 0)
local BLACK = Color3.fromRGB(20, 20, 20)
local GRAY = Color3.fromRGB(42, 42, 42)
local CODE_BG = Color3.fromRGB(30, 30, 30)
local BUTTON = Color3.fromRGB(105, 105, 105)
local BUTTON_HOVER = Color3.fromRGB(125, 125, 125)

local function create(className, properties, parent)
	local instance = Instance.new(className)
	for key, value in pairs(properties) do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

local playerGui = Players.LocalPlayer:WaitForChild("PlayerGui")

-- Remove old copies
for _, container in ipairs({CoreGui, playerGui}) do
	pcall(function()
		local old = container:FindFirstChild("NoticeGui")
		if old then
			old:Destroy()
		end
	end)
end

-- ScreenGui
local gui = create("ScreenGui", {
	Name = "NoticeGui",
	ResetOnSpawn = false,
	IgnoreGuiInset = true,
	ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
})

if not pcall(function() gui.Parent = CoreGui end) then
	gui.Parent = playerGui
end

-- Hazard stripes
local function createHazard(position, direction)
	local holder = create("Frame", {
		Name = "Hazard",
		Size = UDim2.new(1, 0, 0, HAZARD_HEIGHT),
		Position = position,
		BackgroundColor3 = YELLOW,
		BorderSizePixel = 0,
		ClipsDescendants = true,
	}, main)

	local stripWidth = MAIN_WIDTH + PERIOD * 4
	local strip = create("Frame", {
		Name = "Strip",
		Size = UDim2.new(0, stripWidth, 1, 0),
		BackgroundColor3 = YELLOW,
		BorderSizePixel = 0,
	}, holder)

	for i = 0, math.ceil(stripWidth / PERIOD) do
		create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Size = UDim2.new(0, 10, 0, 40),
			Position = UDim2.new(0, i * PERIOD, 0.5, 0),
			Rotation = 45,
			BackgroundColor3 = BLACK,
			BorderSizePixel = 0,
		}, strip)
	end

	local from, to = -PERIOD * 2, -PERIOD
	if direction < 0 then
		from, to = -PERIOD, -PERIOD * 2
	end

	strip.Position = UDim2.new(0, from, 0, 0)
	TweenService:Create(
		strip,
		TweenInfo.new(1.6, Enum.EasingStyle.Linear, Enum.EasingDirection.Out, -1),
		{Position = UDim2.new(0, to, 0, 0)}
	):Play()
end

createHazard(UDim2.new(0, 0, 0, 0), 1)
createHazard(UDim2.new(0, 0, 1, -HAZARD_HEIGHT), -1)

local function createButton(text, order)
	local button = create("TextButton", {
		Name = text,
		Size = UDim2.new(0.3333, -6, 1, 0),
		BackgroundColor3 = BUTTON,
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Font = Enum.Font.GothamBold,
		Text = text,
		TextColor3 = Color3.fromRGB(255, 255, 255),
		TextSize = 13,
		LayoutOrder = order,
	}, buttons)

	button.MouseEnter:Connect(function()
		TweenService:Create(button, TweenInfo.new(0.15), {BackgroundColor3 = BUTTON_HOVER}):Play()
	end)
	button.MouseLeave:Connect(function()
		TweenService:Create(button, TweenInfo.new(0.15), {BackgroundColor3 = BUTTON}):Play()
	end)

	return button
end

local copyButton = createButton("Copy", 1)
local closeButton = createButton("Close", 2)
local executeButton = createButton("Execute", 3)

-- Close animation
local closing = false

local function close()
	if closing then
		return
	end
	closing = true

	local tween = TweenService:Create(
		main,
		TweenInfo.new(1, Enum.EasingStyle.Back, Enum.EasingDirection.In),
		{Position = UDim2.new(0.5, 0, 1, 200)}
	)
	tween.Completed:Connect(function()
		gui:Destroy()
	end)
	tween:Play()
end

-- Button actions
copyButton.MouseButton1Click:Connect(function()
	local clipboard = setclipboard or toclipboard
	local success = clipboard and pcall(clipboard, SCRIPT)

	copyButton.Text = success and "Copied!" or "Failed"
	task.delay(1.2, function()
		if copyButton.Parent then
			copyButton.Text = "Copy"
		end
	end)
end)

closeButton.MouseButton1Click:Connect(close)

executeButton.MouseButton1Click:Connect(function()
	if closing then
		return
	end

	local success, err = pcall(function()
		loadstring(SCRIPT)()
	end)

	if success then
		close()
	else
		warn("Execute failed: " .. tostring(err))
		executeButton.Text = "Error"
		task.delay(1.2, function()
			if executeButton.Parent then
				executeButton.Text = "Execute"
			end
		end)
	end
end)

-- Intro animation
local function slideIn(object, target, delayTime)
	task.delay(delayTime, function()
		TweenService:Create(
			object,
			TweenInfo.new(0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
			{Position = target}
		):Play()
	end)
end

	TweenInfo.new(1.4, Enum.EasingStyle.Bounce, Enum.EasingDirection.Out),
	{Position = UDim2.new(0.5, 0, 0.5, 0)}
):Play()
