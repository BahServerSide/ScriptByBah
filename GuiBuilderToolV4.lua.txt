-- Utils
local BORDERED_CLASSES = {
	Frame = true, TextButton = true, TextLabel = true, TextBox = true,
	ImageLabel = true, ImageButton = true, ScrollingFrame = true,
	VideoFrame = true, CanvasGroup = true,
}

function F.new(className, props, children)
	local inst = Instance.new(className)
	if BORDERED_CLASSES[className] then
		inst.BorderSizePixel = 0
	end
	for prop, value in pairs(props or {}) do
		inst[prop] = value
	end
	for _, child in ipairs(children or {}) do
		child.Parent = inst
	end
	return inst
end

function F.round(number, decimals)
	local mult = 10 ^ (decimals or 0)
	return math.floor(number * mult + 0.5) / mult
end

function F.colorToHex(color)
	return string.format("%02X%02X%02X",
		math.floor(color.R * 255 + 0.5),
		math.floor(color.G * 255 + 0.5),
		math.floor(color.B * 255 + 0.5))
end

function F.hexToColor(hex)
	hex = hex:gsub("#", ""):gsub("%s", "")
	if #hex ~= 6 then return nil end
	local r = tonumber(hex:sub(1, 2), 16)
	local g = tonumber(hex:sub(3, 4), 16)
	local b = tonumber(hex:sub(5, 6), 16)
	if not (r and g and b) then return nil end
	return Color3.fromRGB(r, g, b)
end

-- Inset correction (0 if source ScreenGui also ignores GuiInset)
function F.getInsetCorrection(inst)
	local node = inst
	while node and not node:IsA("ScreenGui") do
		node = node.Parent
	end
	if node and node.IgnoreGuiInset then
		return Vector2.new(0, 0)
	end
	return GuiService:GetGuiInset()
end

-- Drag tool windows
function F.makeDraggable(frame, handle)
	local dragging = false
	local dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

end

-- Shared animated dropdown (list popup)
-- Always opens right below the button; the arrow icon points down and rotates up while open
State.activeDropdownClose = nil
State.activeDropdownAnchor = nil

function F.closeActiveDropdown()
	if State.activeDropdownClose then
		local fn = State.activeDropdownClose
		State.activeDropdownClose = nil
		State.activeDropdownAnchor = nil
		pcall(fn)
	end
end

-- Adds the arrow icon (down) on the right side of a dropdown button
function F.attachDropdownArrow(btn)
	local old = btn:FindFirstChild("DropdownArrow")
	if old then return old end
	local padRight = 24
	local pad = btn:FindFirstChildOfClass("UIPadding")
	if not pad then
		pad = Instance.new("UIPadding")
		pad.Parent = btn
	end
	pad.PaddingRight = UDim.new(0, padRight)
	btn.TextTruncate = Enum.TextTruncate.AtEnd
	-- UIPadding shifts the child area, so the offset compensates to sit 6px from the right edge
	local arrow = F.new("ImageLabel", {
		Name = "DropdownArrow",
		AnchorPoint = Vector2.new(1, 0.5),
		Size = UDim2.fromOffset(14, 14),
		Position = UDim2.new(1, padRight - 6, 0.5, 0),
		BackgroundTransparency = 1,
		Image = ICONS.DropdownArrow,
		ImageColor3 = THEME.Text,
		ScaleType = Enum.ScaleType.Fit,
		Rotation = 0,
		ZIndex = btn.ZIndex + 1,
		Parent = btn,
	})
	return arrow
end

function F.setDropdownArrow(btn, open)
	if not btn then return end
	pcall(function()
		local arrow = btn:FindFirstChild("DropdownArrow")
		if arrow then
			TweenService:Create(arrow, TweenInfo.new(0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
				Rotation = open and 180 or 0,
			}):Play()
		end
	end)
end

function F.isGuiVisible(inst, stopAt)
	local node = inst
	while node and node ~= stopAt do
		if node:IsA("GuiObject") and not node.Visible then
			return false
		end
		node = node.Parent
	end
	return node == stopAt
end

function F.openAnimatedDropdown(anchorBtn, items, currentLabel, onSelect, opts)
	opts = opts or {}

	-- clicking the same button again closes its dropdown
	if State.activeDropdownClose and State.activeDropdownAnchor == anchorBtn then
		F.closeActiveDropdown()
		return
	end
	F.closeActiveDropdown()

	local maxVisible = opts.maxVisible or 6
	local itemH = opts.itemH or 26
	local width = opts.width or math.max(anchorBtn.AbsoluteSize.X, 140)
	local useFontPreview = opts.fontPreview == true

	local n = #items
	local vis = math.min(n, maxVisible)

	local host = screenGui
	local viewport = host.AbsoluteSize
	local abs = anchorBtn.AbsolutePosition
	local asz = anchorBtn.AbsoluteSize
	local px = math.clamp(abs.X, 4, math.max(4, viewport.X - width - 4))
	local py = abs.Y + asz.Y + 4 -- always open below the button

	-- keep it below the button even near the bottom of the screen (shrinks and scrolls)
	local avail = viewport.Y - py - 6
	local listH = math.max(itemH + 6, math.min(vis * itemH + 6, avail))

	local baseTransp = State.uiTransparencyEnabled and (0.25 + State.UI_TRANSPARENCY_AMOUNT * 0.5) or 0.25
	local rowTransp = State.uiTransparencyEnabled and (0.35 + State.UI_TRANSPARENCY_AMOUNT * 0.4) or 0.15

	local drop = F.new("Frame", {
		Name = "__DropdownPopup",
		Size = UDim2.fromOffset(width, 0),
		Position = UDim2.fromOffset(px, py),
		BackgroundColor3 = THEME.Panel,
		BackgroundTransparency = 1, -- start fully transparent
		BorderSizePixel = 0,
		ClipsDescendants = true,
		ZIndex = 200,
		Parent = host,
	})
	F.new("UICorner", { CornerRadius = UDim.new(0, 6), Parent = drop })
	-- no UIStroke (clean transparent look)

	local scroll = F.new("ScrollingFrame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 4,
		CanvasSize = UDim2.new(0, 0, 0, n * itemH),
		ZIndex = 201,
		Parent = drop,
	})
	F.new("UIListLayout", {
		SortOrder = Enum.SortOrder.LayoutOrder,
		Padding = UDim.new(0, 0),
		Parent = scroll,
	})

	local closed = false
	local followConn = nil
	local inputConn = nil

	local function close()
		if closed then return end
		closed = true
		if State.activeDropdownClose == close then
			State.activeDropdownClose = nil
			State.activeDropdownAnchor = nil
		end
		if followConn then followConn:Disconnect() followConn = nil end
		if inputConn then inputConn:Disconnect() inputConn = nil end
		F.setDropdownArrow(anchorBtn, false) -- arrow turns back down
		local tw = TweenService:Create(drop, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
			Size = UDim2.fromOffset(width, 0),
			BackgroundTransparency = 1,
		})
		tw:Play()
		tw.Completed:Connect(function()
			if drop and drop.Parent then drop:Destroy() end
		end)
	end
	State.activeDropdownClose = close
	State.activeDropdownAnchor = anchorBtn

	for i, item in ipairs(items) do
		local label = item
		local fontForItem = nil
		if type(item) == "table" then
			label = item.Label or item.Name or tostring(item.Value)
			fontForItem = item.Font
		end
		local isCurrent = (tostring(label) == tostring(currentLabel))
		local btn = F.new("TextButton", {
			Size = UDim2.new(1, 0, 0, itemH),
			BackgroundColor3 = isCurrent and THEME.RowSelected or THEME.PanelAlt,
			BackgroundTransparency = 1, -- fade in with list
			AutoButtonColor = false,
			Font = (useFontPreview and fontForItem) or Enum.Font.Gotham,
			Text = "  " .. tostring(label),
			TextColor3 = THEME.Text,
			TextTransparency = 1,
			TextSize = 12,
			TextXAlignment = Enum.TextXAlignment.Left,
			LayoutOrder = i,
			ZIndex = 202,
			Parent = scroll,
		})
		btn.MouseEnter:Connect(function()
			if not isCurrent then
				btn.BackgroundColor3 = THEME.ButtonHover
			end
		end)
		btn.MouseLeave:Connect(function()
			btn.BackgroundColor3 = isCurrent and THEME.RowSelected or THEME.PanelAlt
		end)
		btn.MouseButton1Click:Connect(function()
			onSelect(item, label)
			close()
		end)
		-- slow fade-in for each row
		TweenService:Create(btn, TweenInfo.new(0.45, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			BackgroundTransparency = rowTransp,
			TextTransparency = 0,
		}):Play()
	end

	-- slow expand + fade to target transparency
	TweenService:Create(drop, TweenInfo.new(0.45, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
		Size = UDim2.fromOffset(width, listH),
		BackgroundTransparency = baseTransp,
	}):Play()

	-- arrow turns up while the dropdown is open
	F.setDropdownArrow(anchorBtn, true)

	-- stay glued under the button (follows window drag); close if the button is gone/hidden
	followConn = RunService.RenderStepped:Connect(function()
		if closed then return end
		if not anchorBtn.Parent or not F.isGuiVisible(anchorBtn, host) then
			close()
			return
		end
		local a = anchorBtn.AbsolutePosition
		local s = anchorBtn.AbsoluteSize
		local vp = host.AbsoluteSize
		local x = math.clamp(a.X, 4, math.max(4, vp.X - width - 4))
		drop.Position = UDim2.fromOffset(x, a.Y + s.Y + 4)
	end)

	task.defer(function()
		if closed then return end
		inputConn = UserInputService.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				local pos = input.Position
				local dpos = drop.AbsolutePosition
				local dsz = drop.AbsoluteSize
				local insideDrop = pos.X >= dpos.X and pos.X <= dpos.X + dsz.X
					and pos.Y >= dpos.Y and pos.Y <= dpos.Y + dsz.Y
				local bpos = anchorBtn.AbsolutePosition
				local bsz = anchorBtn.AbsoluteSize
				local insideBtn = pos.X >= bpos.X and pos.X <= bpos.X + bsz.X
					and pos.Y >= bpos.Y and pos.Y <= bpos.Y + bsz.Y
				if not insideDrop and not insideBtn then
					close()
				end
			end
		end)
	end)

	return close
end

function F.luaValue(value)
	local t = typeof(value)
	if t == "string" then
		return string.format("%q", value)
	elseif t == "boolean" or t == "number" then
		return tostring(value)
	elseif t == "UDim2" then
		return string.format("UDim2.new(%s, %s, %s, %s)",
			F.round(value.X.Scale, 4), value.X.Offset, F.round(value.Y.Scale, 4), value.Y.Offset)
	elseif t == "UDim" then
		return string.format("UDim.new(%s, %s)", F.round(value.Scale, 4), value.Offset)
	elseif t == "Color3" then
		return string.format("Color3.fromRGB(%d, %d, %d)",
			math.floor(value.R * 255 + 0.5), math.floor(value.G * 255 + 0.5), math.floor(value.B * 255 + 0.5))
	elseif t == "Vector2" then
		return string.format("Vector2.new(%s, %s)", F.round(value.X, 4), F.round(value.Y, 4))
	elseif t == "EnumItem" then
		return "Enum." .. tostring(value.EnumType) .. "." .. value.Name
	elseif t == "ColorSequence" then
		local parts = {}
		for _, kp in ipairs(value.Keypoints) do
			table.insert(parts, string.format("ColorSequenceKeypoint.new(%s, %s)",
				F.round(kp.Time, 4), F.luaValue(kp.Value)))
		end
		return "ColorSequence.new({" .. table.concat(parts, ", ") .. "})"
	elseif t == "NumberSequence" then
		local parts = {}
		for _, kp in ipairs(value.Keypoints) do
			if kp.Envelope and kp.Envelope ~= 0 then
				table.insert(parts, string.format("NumberSequenceKeypoint.new(%s, %s, %s)",
					F.round(kp.Time, 4), F.round(kp.Value, 4), F.round(kp.Envelope, 4)))
			else
				table.insert(parts, string.format("NumberSequenceKeypoint.new(%s, %s)",
					F.round(kp.Time, 4), F.round(kp.Value, 4)))
			end
		end
		return "NumberSequence.new({" .. table.concat(parts, ", ") .. "})"
	elseif t == "Rect" then
		return string.format("Rect.new(%s, %s, %s, %s)",
			F.round(value.Min.X, 4), F.round(value.Min.Y, 4), F.round(value.Max.X, 4), F.round(value.Max.Y, 4))
	elseif t == "Font" then
		return string.format("Font.new(%q, Enum.FontWeight.%s, Enum.FontStyle.%s)",
			value.Family, value.Weight.Name, value.Style.Name)
	else
		return tostring(value)
	end
end

function F.clearBBConnections()
	for _, c in ipairs(State.bbConnections) do
		c:Disconnect()
	end
	State.bbConnections = {}
end

function F.destroyBoundingBox()
	F.clearBBConnections()
	State.isDraggingBB = false
	State.isResizingBB = false
	State.resizeMode = nil
	if State.boundingBoxFrame then
		State.boundingBoxFrame:Destroy()
		State.boundingBoxFrame = nil
	end
	State.boundingHandles = {}
end

function F.getParentAbs(parent)
	if parent and parent:IsA("GuiObject") then
		return parent.AbsolutePosition, parent.AbsoluteSize
	end
	local cam = Workspace.CurrentCamera
	local vs = cam and cam.ViewportSize or Vector2.new(1280, 720)
	local origin = Vector2.new(0, 0)
	if parent and parent:IsA("LayerCollector") and not parent.IgnoreGuiInset then
		origin = GuiService:GetGuiInset()
	end
	return origin, vs
end

function F.makeAlignLine(isVertical, pos, length, offset, inset)
	-- Same inset issue as the bounding box
				inset = inset or GuiService:GetGuiInset()
	local line = F.new("Frame", {
		Name = "__AlignLine",
		BackgroundColor3 = Color3.fromRGB(255, 80, 180),
		BackgroundTransparency = 0.25,
		BorderSizePixel = 0,
		ZIndex = 9998,
		Parent = UI.alignmentFolder,
	})
	if isVertical then
		line.Size = UDim2.fromOffset(1, length)
		line.Position = UDim2.fromOffset(pos + inset.X, offset + inset.Y)
	else
		line.Size = UDim2.fromOffset(length, 1)
		line.Position = UDim2.fromOffset(offset + inset.X, pos + inset.Y)
	end
	table.insert(State.alignmentLines, line)
	return line
end

-- Snap against parent and sibling edges
function F.snapToGuides(inst, proposedAbsPos, proposedAbsSize)
	if not State.alignmentEnabled or not inst or not inst:IsA("GuiObject") then
		return proposedAbsPos, proposedAbsSize, {}
	end
	F.clearAlignmentLines()

	local inset = F.getInsetCorrection(inst)
	local parent = inst.Parent
	local parentPos, parentSize
	if parent and parent:IsA("GuiObject") then
		parentPos = parent.AbsolutePosition
		parentSize = parent.AbsoluteSize
	else
		local cam = Workspace.CurrentCamera
		local vs = cam and cam.ViewportSize or Vector2.new(1280, 720)
		parentPos = Vector2.new(0, 0)
		parentSize = vs
	end

	local px, py = proposedAbsPos.X, proposedAbsPos.Y
	local w, h = proposedAbsSize.X, proposedAbsSize.Y
	local snappedX, snappedY = px, py
	local guides = {}

	-- Parent reference points
	local refsX = {
		{ pos = parentPos.X, kind = "left" },
		{ pos = parentPos.X + parentSize.X / 2, kind = "center" },
		{ pos = parentPos.X + parentSize.X, kind = "right" },
	}
	local refsY = {
		{ pos = parentPos.Y, kind = "top" },
		{ pos = parentPos.Y + parentSize.Y / 2, kind = "center" },
		{ pos = parentPos.Y + parentSize.Y, kind = "bottom" },
	}

	-- Element's own edges
	local function trySnapX(edgeAbs)
		for _, ref in ipairs(refsX) do
			if math.abs(edgeAbs - ref.pos) <= State.snapThreshold then
				return ref.pos, ref
			end
		end
		return nil, nil
	end
	local function trySnapY(edgeAbs)
		for _, ref in ipairs(refsY) do
			if math.abs(edgeAbs - ref.pos) <= State.snapThreshold then
				return ref.pos, ref
			end
		end
		return nil, nil
	end

	-- left
	local s, ref = trySnapX(px)
	if s then
		snappedX = s
		table.insert(guides, { vertical = true, pos = s })
	else
		-- center
		s, ref = trySnapX(px + w / 2)
		if s then
			snappedX = s - w / 2
			table.insert(guides, { vertical = true, pos = s })
		else
			-- right
			s, ref = trySnapX(px + w)
			if s then
				snappedX = s - w
				table.insert(guides, { vertical = true, pos = s })
			end
		end
	end

	s, ref = trySnapY(py)
	if s then
		snappedY = s
		table.insert(guides, { vertical = false, pos = s })
	else
		s, ref = trySnapY(py + h / 2)
		if s then
			snappedY = s - h / 2
			table.insert(guides, { vertical = false, pos = s })
		else
			s, ref = trySnapY(py + h)
			if s then
				snappedY = s - h
				table.insert(guides, { vertical = false, pos = s })
			end
		end
	end

	-- Snap against siblings
	if parent then
		for _, sib in ipairs(parent:GetChildren()) do
			if sib ~= inst and sib:IsA("GuiObject") and not F.isOverlay(sib) then
				local sx, sy = sib.AbsolutePosition.X, sib.AbsolutePosition.Y
				local sw, sh = sib.AbsoluteSize.X, sib.AbsoluteSize.Y
				local sibRefsX = { sx, sx + sw / 2, sx + sw }
				local sibRefsY = { sy, sy + sh / 2, sy + sh }
				for _, rp in ipairs(sibRefsX) do
					if math.abs(snappedX - rp) <= State.snapThreshold then
						snappedX = rp
						table.insert(guides, { vertical = true, pos = rp })
					elseif math.abs(snappedX + w / 2 - rp) <= State.snapThreshold then
						snappedX = rp - w / 2
						table.insert(guides, { vertical = true, pos = rp })
					elseif math.abs(snappedX + w - rp) <= State.snapThreshold then
						snappedX = rp - w
						table.insert(guides, { vertical = true, pos = rp })
					end
				end
				for _, rp in ipairs(sibRefsY) do
					if math.abs(snappedY - rp) <= State.snapThreshold then
						snappedY = rp
						table.insert(guides, { vertical = false, pos = rp })
					elseif math.abs(snappedY + h / 2 - rp) <= State.snapThreshold then
						snappedY = rp - h / 2
						table.insert(guides, { vertical = false, pos = rp })
					elseif math.abs(snappedY + h - rp) <= State.snapThreshold then
						snappedY = rp - h
						table.insert(guides, { vertical = false, pos = rp })
					end
				end
			end
		end
	end

	-- Draw guide lines
	for _, g in ipairs(guides) do
		if g.vertical then
			F.makeAlignLine(true, g.pos, parentSize.Y, parentPos.Y, inset)
		else
			F.makeAlignLine(false, g.pos, parentSize.X, parentPos.X, inset)
		end
	end

	return Vector2.new(snappedX, snappedY), proposedAbsSize, guides
end

function F.applySnappedPosition(inst, absPos, absSize)
	local parentPos = F.getParentAbs(inst.Parent)
	local relX = absPos.X - parentPos.X
	local relY = absPos.Y - parentPos.Y
	inst.Position = UDim2.new(0, relX, 0, relY)
	inst.Size = UDim2.new(0, absSize.X, 0, absSize.Y)
end


function F.updateBoundingBox()
	if not State.selected or not State.selected:IsA("GuiObject") or not State.boundingBoxFrame then return end
	if not State.selected.Parent then
		F.destroyBoundingBox()
		return
	end

	-- Mirror in the SAME parent (avoids GuiInset teleport)
	if State.boundingBoxFrame.Parent ~= State.selected.Parent then
		State.boundingBoxFrame.Parent = State.selected.Parent
	end
	State.boundingBoxFrame.AnchorPoint = State.selected.AnchorPoint
	State.boundingBoxFrame.Position = State.selected.Position
	State.boundingBoxFrame.Size = State.selected.Size
	State.boundingBoxFrame.Rotation = State.selected.Rotation
	State.boundingBoxFrame.ZIndex = (tonumber(State.selected.ZIndex) or 1) + 50

	local hs = State.HANDLE_SIZE
	local positions = {
		TL = UDim2.new(0, -hs/2, 0, -hs/2),
		TR = UDim2.new(1, -hs/2, 0, -hs/2),
		BL = UDim2.new(0, -hs/2, 1, -hs/2),
		BR = UDim2.new(1, -hs/2, 1, -hs/2),
		T  = UDim2.new(0.5, -hs/2, 0, -hs/2),
		B  = UDim2.new(0.5, -hs/2, 1, -hs/2),
		L  = UDim2.new(0, -hs/2, 0.5, -hs/2),
		R  = UDim2.new(1, -hs/2, 0.5, -hs/2),
	}
	for name, handle in pairs(State.boundingHandles) do
		if handle and handle.Parent then
			handle.Position = positions[name]
			handle.ZIndex = State.boundingBoxFrame.ZIndex + 1
		end
	end
end

function F.createBoundingBox(inst)
	F.destroyBoundingBox()
	if not inst or not inst:IsA("GuiObject") then return end

	-- Fully transparent frame (outline only)
	State.boundingBoxFrame = F.new("Frame", {
		Name = "__BoundingBox",
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 10000,
		Active = true,
		Parent = inst.Parent or screenGui,
	})

	F.new("UIStroke", {
		Color = State.ORANGE,
		Thickness = 2,
		Transparency = 0.6, -- opacidade 0.40
		Parent = State.boundingBoxFrame,
	})

	local function makeHandle(name)
		local h = F.new("Frame", {
			Name = "__Handle_" .. name,
			Size = UDim2.fromOffset(State.HANDLE_SIZE, State.HANDLE_SIZE),
			BackgroundColor3 = State.ORANGE,
			BackgroundTransparency = 0.6, -- opacidade 0.40, sem contorno (sem UIStroke)
			BorderSizePixel = 0,
			ZIndex = 10001,
			Active = true,
			Parent = State.boundingBoxFrame,
		})
		F.new("UICorner", { CornerRadius = UDim.new(1, 0), Parent = h })
		State.boundingHandles[name] = h
		return h
	end

	local hTL = makeHandle("TL")
	local hTR = makeHandle("TR")
	local hBL = makeHandle("BL")
	local hBR = makeHandle("BR")
	local hT  = makeHandle("T")
	local hB  = makeHandle("B")
	local hL  = makeHandle("L")
	local hR  = makeHandle("R")

	F.updateBoundingBox()

	local function startDrag(input)
		if not State.selected or not State.selected:IsA("GuiObject") then return end
		State.isDraggingBB = true
		State.isResizingBB = false
		State.resizeMode = nil
		State.dragStartPos = input.Position
		State.startUDimPos = State.selected.Position
		State.startAbsPos = State.selected.AbsolutePosition
		State.startAbsSize = State.selected.AbsoluteSize
	end

	local function doDrag(input)
		if not State.isDraggingBB or not State.selected then return end
		local delta = input.Position - State.dragStartPos
		-- Proposed abs position from mouse delta
						local proposedAbsPos = Vector2.new(State.startAbsPos.X + delta.X, State.startAbsPos.Y + delta.Y)
		local snappedAbsPos = F.snapToGuides(State.selected, proposedAbsPos, State.startAbsSize)
		local finalDelta = Vector2.new(snappedAbsPos.X - State.startAbsPos.X, snappedAbsPos.Y - State.startAbsPos.Y)
		-- Direct delta on Position (works anywhere)
		State.selected.Position = UDim2.new(
			State.startUDimPos.X.Scale,
			State.startUDimPos.X.Offset + finalDelta.X,
			State.startUDimPos.Y.Scale,
			State.startUDimPos.Y.Offset + finalDelta.Y
		)
		F.updateBoundingBox()
	end

	local function startResize(input, mode)
		if not State.selected or not State.selected:IsA("GuiObject") then return end
		State.isResizingBB = true
		State.isDraggingBB = false
		State.resizeMode = mode
		State.dragStartPos = input.Position
		State.startAbsPos = State.selected.AbsolutePosition
		State.startAbsSize = State.selected.AbsoluteSize
		State.startUDimPos = State.selected.Position
		State.startUDimSize = State.selected.Size
	end

	local function doResize(input)
		if not State.isResizingBB or not State.selected or not State.resizeMode then return end
		local delta = input.Position - State.dragStartPos
		local ap = State.selected.AnchorPoint
		local newW = State.startAbsSize.X
		local newH = State.startAbsSize.Y

		local moveL = (State.resizeMode == "L" or State.resizeMode == "TL" or State.resizeMode == "BL")
		local moveR = (State.resizeMode == "R" or State.resizeMode == "TR" or State.resizeMode == "BR")
		local moveT = (State.resizeMode == "T" or State.resizeMode == "TL" or State.resizeMode == "TR")
		local moveB = (State.resizeMode == "B" or State.resizeMode == "BL" or State.resizeMode == "BR")

		if moveR then newW = math.max(16, State.startAbsSize.X + delta.X) end
		if moveL then newW = math.max(16, State.startAbsSize.X - delta.X) end
		if moveB then newH = math.max(16, State.startAbsSize.Y + delta.Y) end
		if moveT then newH = math.max(16, State.startAbsSize.Y - delta.Y) end

		-- Keep opposite edge fixed
		local ox, oy = 0, 0
		if moveL and not moveR then
			ox = (State.startAbsSize.X - newW) * (1 - ap.X)
		elseif moveR and not moveL then
			ox = (newW - State.startAbsSize.X) * ap.X
		end
		if moveT and not moveB then
			oy = (State.startAbsSize.Y - newH) * (1 - ap.Y)
		elseif moveB and not moveT then
			oy = (newH - State.startAbsSize.Y) * ap.Y
		end

		State.selected.Position = UDim2.new(
			State.startUDimPos.X.Scale, State.startUDimPos.X.Offset + ox,
			State.startUDimPos.Y.Scale, State.startUDimPos.Y.Offset + oy
		)
		State.selected.Size = UDim2.fromOffset(newW, newH)
		F.updateBoundingBox()
	end

	local function endInteraction()
		State.isDraggingBB = false
		State.isResizingBB = false
		State.resizeMode = nil
		F.clearAlignmentLines()
		F.refreshProperties()
	end

	-- Drag body (move)
	table.insert(State.bbConnections, State.boundingBoxFrame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			startDrag(input)
			local conn
			conn = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					endInteraction()
					conn:Disconnect()
				end
			end)
		end
	end))

	local function connectHandle(handle, mode)
		table.insert(State.bbConnections, handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				startResize(input, mode)
				local conn
				conn = input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then
						endInteraction()
						conn:Disconnect()
					end
				end)
			end
		end))
	end

	connectHandle(hTL, "TL")
	connectHandle(hTR, "TR")
	connectHandle(hBL, "BL")
	connectHandle(hBR, "BR")
	connectHandle(hT, "T")
	connectHandle(hB, "B")
	connectHandle(hL, "L")
	connectHandle(hR, "R")

	table.insert(State.bbConnections, UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			if State.isDraggingBB then
				doDrag(input)
			elseif State.isResizingBB then
				doResize(input)
			end
		end
	end))

	table.insert(State.bbConnections, RunService.RenderStepped:Connect(function()
		if State.selected and State.selected:IsA("GuiObject") and State.selected.Parent and State.boundingBoxFrame then
			if not State.isDraggingBB and not State.isResizingBB then
				F.updateBoundingBox()
			end
		end
	end))
end

function F.repositionAddElementPopup()
	-- Follows the Explorer window
		local viewport = Workspace.CurrentCamera and Workspace.CurrentCamera.ViewportSize or Vector2.new(1280, 720)
	local popupWidth = UI.addElementPopup.Size.X.Offset
	local popupHeight = UI.addElementPopup.Size.Y.Offset
	local x = UI.explorerWindow.AbsolutePosition.X + UI.explorerWindow.AbsoluteSize.X + 6
	if x + popupWidth > viewport.X then
		x = UI.explorerWindow.AbsolutePosition.X - popupWidth - 6
	end
	local y = math.clamp(UI.explorerWindow.AbsolutePosition.Y, 0, math.max(viewport.Y - popupHeight, 0))
	UI.addElementPopup.Position = UDim2.new(0, x, 0, y)
end

State.FADE_INFO = TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
State.FADE_INFO_SLOW = TweenInfo.new(0.55, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

function F.getTargetWindowTransparency()
	return State.uiTransparencyEnabled and State.UI_TRANSPARENCY_AMOUNT or 0
end

function F.getTargetPanelTransparency()
	-- rows / top bars slightly more transparent than main window when mode is on
	if State.uiTransparencyEnabled then
		return math.clamp(State.UI_TRANSPARENCY_AMOUNT + 0.12, 0, 0.85)
	end
	return 0
end

function F.getTargetOptionsTransparency()
	if State.uiTransparencyEnabled then
		return math.clamp(0.19 + State.UI_TRANSPARENCY_AMOUNT * 0.6, 0.19, 0.75)
	end
	return 0.19
end

function F.tweenBgTransparency(inst, target, info)
	if not inst then return end
	info = info or State.FADE_INFO
	pcall(function()
		TweenService:Create(inst, info, { BackgroundTransparency = target }):Play()
	end)
end

function F.applyTransparencyToWindowTree(win, mainT, panelT, info)
	if not win then return end
	info = info or State.FADE_INFO
	F.tweenBgTransparency(win, mainT, info)
	for _, child in ipairs(win:GetDescendants()) do
		if child:IsA("GuiObject") and not F.isOverlay(child) then
			local name = child.Name
			-- tops, rows, action bars, list backgrounds
			if name == "ActionBar" or name:find("Top") or name == "Row"
				or child.Parent and child.Parent.Name == "ExplorerList"
				or child.Parent and child.Parent.Name == "PropertiesList" then
				if child:IsA("Frame") or child:IsA("TextButton") then
					-- don't force fully transparent text labels
					if child.BackgroundTransparency < 0.99 then
						F.tweenBgTransparency(child, panelT, info)
					end
				end
			end
		end
	end
	-- common direct children named frames
	for _, child in ipairs(win:GetChildren()) do
		if child:IsA("Frame") and child.BackgroundTransparency < 0.99 then
			if child.Name:find("Top") or child.Name == "ActionBar" then
				F.tweenBgTransparency(child, panelT, info)
			end
		end
	end
end

function F.applyUiTransparency(enabled)
	State.uiTransparencyEnabled = enabled
	local mainT = F.getTargetWindowTransparency()
	local panelT = F.getTargetPanelTransparency()
	local optT = F.getTargetOptionsTransparency()
	local info = State.FADE_INFO_SLOW

	local windows = {
		UI.explorerWindow, UI.propertiesWindow, UI.settingsWindow, UI.saveLoadWindow,
		UI.gui2luaWindow, UI.codeOutputWindow, UI.sourceEditorWindow,
	}
	for _, w in ipairs(windows) do
		F.applyTransparencyToWindowTree(w, mainT, panelT, info)
	end

	if UI.optionsMenu then
		F.tweenBgTransparency(UI.optionsMenu, optT, info)
		for _, child in ipairs(UI.optionsMenu:GetDescendants()) do
			if child:IsA("TextButton") and child.BackgroundTransparency < 0.99 then
				F.tweenBgTransparency(child, math.clamp(optT + 0.08, 0, 0.85), info)
			end
		end
	end
	if UI.optionsButton then
		F.tweenBgTransparency(UI.optionsButton, optT, info)
	end
end

-- Fade-open any tool window: starts at transparency 1, tweens to current target
function F.showWindowWithFade(win)
	if not win then return end
	local mainT = F.getTargetWindowTransparency()
	local panelT = F.getTargetPanelTransparency()
	win.Visible = true
	-- start invisible
	pcall(function() win.BackgroundTransparency = 1 end)
	for _, child in ipairs(win:GetChildren()) do
		if child:IsA("Frame") and child.BackgroundTransparency < 0.99 then
			pcall(function() child.BackgroundTransparency = 1 end)
		end
	end
	F.applyTransparencyToWindowTree(win, mainT, panelT, State.FADE_INFO_SLOW)
end

function F.hideWindowWithFade(win)
	if not win or not win.Visible then return end
	local info = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
	local tw = TweenService:Create(win, info, { BackgroundTransparency = 1 })
	tw:Play()
	for _, child in ipairs(win:GetChildren()) do
		if child:IsA("Frame") then
			TweenService:Create(child, info, { BackgroundTransparency = 1 }):Play()
		end
	end
	tw.Completed:Connect(function()
		win.Visible = false
		-- restore target so next open starts clean
		local mainT = F.getTargetWindowTransparency()
		local panelT = F.getTargetPanelTransparency()
		F.applyTransparencyToWindowTree(win, mainT, panelT, TweenInfo.new(0))
	end)
end

function F.toggleWindowWithFade(win)
	if not win then return end
	if win.Visible then
		F.hideWindowWithFade(win)
	else
		F.showWindowWithFade(win)
	end
end


function F.saveSettingsToDisk()
	local data = string.format(
		"theme=%s\nhighlight=%s\nshowHighlight=%s\nshowName=%s\nalign=%s\nuiTransp=%s\n",
		State.currentThemeName,
		F.colorToHex(State.highlightColor),
		tostring(State.showSelectionHighlight),
		tostring(State.showSelectedName),
		tostring(State.alignmentEnabled),
		tostring(State.uiTransparencyEnabled)
	)
	if typeof(writefile) == "function" then
		pcall(writefile, State.SETTINGS_SAVE_KEY .. ".txt", data)
	end
	-- also Attribute on tool ScreenGui
	pcall(function()
		screenGui:SetAttribute("SavedSettings", data)
	end)
end

function F.loadSettingsFromDisk()
	local data = nil
	if typeof(readfile) == "function" and typeof(isfile) == "function" then
		local ok, exists = pcall(isfile, State.SETTINGS_SAVE_KEY .. ".txt")
		if ok and exists then
			local ok2, content = pcall(readfile, State.SETTINGS_SAVE_KEY .. ".txt")
			if ok2 then data = content end
		end
	end
	if not data then
		local attr = screenGui:GetAttribute("SavedSettings")
		if typeof(attr) == "string" then data = attr end
	end
	if not data then return end
	local map = {}
	for line in string.gmatch(data, "[^\n]+") do
		local k, v = line:match("^([^=]+)=(.*)$")
		if k then map[k] = v end
	end
	if map.theme and THEMES[map.theme] then
		State.currentThemeName = map.theme
	end
	if map.highlight then
		local c = F.hexToColor(map.highlight)
		if c then State.highlightColor = c; State.ORANGE = c end
	end
	if map.showHighlight ~= nil then State.showSelectionHighlight = (map.showHighlight == "true") end
	if map.showName ~= nil then State.showSelectedName = (map.showName == "true") end
	if map.align ~= nil then State.alignmentEnabled = (map.align == "true") end
	if map.uiTransp ~= nil then State.uiTransparencyEnabled = (map.uiTransp == "true") end
end

function F.ensureDiskFolder()
	if State.canUseDisk and makefolder and isfolder then
		local ok, exists = pcall(isfolder, State.DISK_SAVE_FOLDER)
		if ok and not exists then
			pcall(makefolder, State.DISK_SAVE_FOLDER)
		end

end

function F.diskSavePath(username, saveName)
	local safeUser = username:gsub("[^%w_%-]", "_")
	local safeName = saveName:gsub("[^%w_%-]", "_")
	return State.DISK_SAVE_FOLDER .. "/" .. safeUser .. "__" .. safeName .. ".txt"
end

-- Serialize instances to Lua source (like Gui2Lua)
function F.serializeTreeToLua(instances)
	local lines = {}
	local varCount = 0
	local function nextVar(name)
		varCount = varCount + 1
		local safe = name:gsub("[^%w_]", "")
		if safe == "" then safe = "Obj" end
		return safe .. varCount
	end
	table.insert(lines, "local __Root = Instance.new(\"Folder\")")
	local function exportInstance(inst, parentVar)
		if F.isOverlay(inst) then return end
		local className = inst.ClassName
		local props = EXPORTABLE_PROPS[className]
		local varName = nextVar(inst.Name)
		table.insert(lines, string.format("local %s = Instance.new(%q)", varName, className))
		if props then
			for _, propName in ipairs(props) do
				local ok, value = pcall(function() return inst[propName] end)
				if ok and value ~= nil then
					table.insert(lines, string.format("%s.%s = %s", varName, propName, F.luaValue(value)))
				end
			end
		else
			table.insert(lines, string.format("%s.Name = %q", varName, inst.Name))
		end
		table.insert(lines, string.format("%s.Parent = %s", varName, parentVar))
		for _, child in ipairs(inst:GetChildren()) do
			exportInstance(child, varName)
		end
	end
	for _, inst in ipairs(instances) do
		exportInstance(inst, "__Root")
	end
	table.insert(lines, "return __Root")
	return table.concat(lines, "\n")
end

-- List saves: disk first, then session
function F.listUserSaves()
	local names = {}
	local seen = {}

	local function addName(saveName)
		if saveName and saveName ~= "" and not seen[saveName] and not saveName:match("^__") then
			seen[saveName] = true
			table.insert(names, saveName)
		end
	end

	if State.canUseDisk and typeof(listfiles) == "function" then
		F.ensureDiskFolder()
		local ok, files = pcall(listfiles, State.DISK_SAVE_FOLDER)
		if ok and files then
			local prefix = LocalPlayer.Name:gsub("[^%w_%-]", "_") .. "__"
			for _, filePath in ipairs(files) do
				local fileName = tostring(filePath):match("([^/\\]+)$") or tostring(filePath)
				if fileName:sub(1, #prefix) == prefix then
					addName(fileName:sub(#prefix + 1):gsub("%.txt$", ""))
				end
			end
		end
	end

	if State.saveRemote then
		local ok, remoteNames = pcall(function()
			return State.saveRemote:InvokeServer("List")
		end)
		if ok and type(remoteNames) == "table" then
			for _, n in ipairs(remoteNames) do
				addName(n)
			end
		end
	end

	local folder = F.getUserSavesFolder()
	for _, child in ipairs(folder:GetChildren()) do
		if child:IsA("Folder") then
			addName(child.Name)
		end
	end

	table.sort(names)
	return names
end

function F.captureProjectSnapshot()
	local clones = {}
	if not State.projectGui then return clones end
	for _, child in ipairs(State.projectGui:GetChildren()) do
		if not F.isOverlay(child) then
			table.insert(clones, child:Clone())
		end
	end
	return clones
end

function F.writeSave(saveName, clones)
	-- session copy (fast)
	local folder = F.getUserSavesFolder()
	local existing = folder:FindFirstChild(saveName)
	if existing then existing:Destroy() end
	local saveFolder = Instance.new("Folder")
	saveFolder.Name = saveName
	saveFolder.Parent = folder
	for _, c in ipairs(clones) do
		c.Parent = saveFolder
	end

	-- Persistência real: disco (executor) + DataStore (servidor)
	local okSrc, srcCode = pcall(serializeTreeToLua, clones)
	if okSrc and srcCode then
		if State.canUseDisk then
			F.ensureDiskFolder()
			pcall(writefile, F.diskSavePath(LocalPlayer.Name, saveName), srcCode)
		end
		if State.saveRemote then
			task.spawn(function()
				pcall(function()
					State.saveRemote:InvokeServer("Save", saveName, srcCode)
				end)
			end)
		end
	end

	return saveFolder
end

function F.loadSave(saveName)
	local folder = F.getUserSavesFolder()
	local saveFolder = folder:FindFirstChild(saveName)
	local sourceInstances = nil
	local tempRoot = nil

	if saveFolder then
		sourceInstances = saveFolder:GetChildren()
	else
		local srcCode = nil
		if State.canUseDisk then
			local path = F.diskSavePath(LocalPlayer.Name, saveName)
			local ok, exists = pcall(isfile, path)
			if ok and exists then
				local ok2, content = pcall(readfile, path)
				if ok2 then srcCode = content end
			end
		end
		if (not srcCode or srcCode == "") and State.saveRemote then
			local okR, content = pcall(function()
				return State.saveRemote:InvokeServer("Load", saveName)
			end)
			if okR and type(content) == "string" and content ~= "" then
				srcCode = content
			end
		end
		if srcCode and srcCode ~= "" then
			local chunk = loadstring(srcCode)
			if chunk then
				local ok3, root = pcall(chunk)
				if ok3 and root then
					tempRoot = root
					sourceInstances = root:GetChildren()
				end
			end
		end
	end

	if not sourceInstances then return false end

	State.selected = nil
	F.destroyBoundingBox()
	F.clearAlignmentLines()
	for _, child in ipairs(State.projectGui:GetChildren()) do
		if not F.isOverlay(child) then
			child:Destroy()
		end
	end
	for _, inst in ipairs(sourceInstances) do
		inst:Clone().Parent = State.projectGui
	end
	if tempRoot then
		tempRoot:Destroy()
	end
	F.refreshExplorer()
	F.selectInstance(State.projectGui)
	return true
end

function F.deleteSave(saveName)
	local removedSomething = false
	local folder = F.getUserSavesFolder()
	local saveFolder = folder:FindFirstChild(saveName)
	if saveFolder then
		saveFolder:Destroy()
		removedSomething = true
	end
	if State.canUseDisk then
		local path = F.diskSavePath(LocalPlayer.Name, saveName)
		local ok, exists = pcall(isfile, path)
		if ok and exists and typeof(delfile) == "function" then
			pcall(delfile, path)
			removedSomething = true
		end
	end
	if State.saveRemote then
		local ok = pcall(function()
			return State.saveRemote:InvokeServer("Delete", saveName)
		end)
		if ok then removedSomething = true end
	end
	return removedSomething
end

-- AUTOSAVE: keeps the active project (not the named saves list) on disk,
-- so it's still there after leaving/rejoining the game.
function F.autosavePath()
	local safeUser = LocalPlayer.Name:gsub("[^%w_%-]", "_")
	return State.DISK_SAVE_FOLDER .. "/__autosave_" .. safeUser .. ".txt"
end

function F.doAutosave()
	if not State.canUseDisk then return end
	F.ensureDiskFolder()
	local clones = F.captureProjectSnapshot()
	local ok, src = pcall(serializeTreeToLua, clones)
	for _, c in ipairs(clones) do
		c:Destroy()
	end
	if ok then
		pcall(writefile, F.autosavePath(), src)
	end
end

function State.loadAutosaveIfExists()
	if not State.canUseDisk then return false end
	local path = F.autosavePath()
	local ok, exists = pcall(isfile, path)
	if not (ok and exists) then return false end
	local ok2, src = pcall(readfile, path)
	if not (ok2 and src) or src == "" then return false end
	local chunk = loadstring(src)
	if not chunk then return false end
	local ok3, root = pcall(chunk)
	if not (ok3 and root) then return false end
	for _, child in ipairs(State.projectGui:GetChildren()) do
		if not F.isOverlay(child) then
			child:Destroy()
		end
	end
	for _, inst in ipairs(root:GetChildren()) do
		inst:Clone().Parent = State.projectGui
	end
	root:Destroy()
	return true
end

-- Autosaves every few seconds while the tool is open (simpler and safer
-- than trying to hook every single edit path).
task.spawn(function()
	while true do
		task.wait(6)
		pcall(doAutosave)
	end
end)

State.refreshSavesList = function()
	for _, child in ipairs(UI.savesScroll:GetChildren()) do
		if child:IsA("Frame") then child:Destroy() end
	end

	local names = F.listUserSaves()
	local order = 0
	for _, saveName in ipairs(names) do
		if filter == "" or saveName:lower():find(filter, 1, true) then
			order = order + 1
			local row = F.new("Frame", {
				Size = UDim2.new(1, -8, 0, 34),
				BackgroundColor3 = THEME.Panel,
				LayoutOrder = order,
				ZIndex = 16,
				Parent = UI.savesScroll,
			})
			F.new("UICorner", { CornerRadius = UDim.new(0, 6), Parent = row })
			F.new("TextLabel", {
				Size = UDim2.new(1, -120, 1, 0),
				Position = UDim2.new(0, 8, 0, 0),
				BackgroundTransparency = 1,
				Font = Enum.Font.Gotham,
				Text = saveName,
				TextColor3 = THEME.Text,
				TextSize = 13,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
				ZIndex = 16,
				Parent = row,
			})
			-- Edit (orange)
			local editBtn = F.makeSaveActionBtn(row, ICONS.EditSave, Color3.fromRGB(200, 180, 40), -116)
			editBtn.MouseButton1Click:Connect(function()
				if F.loadSave(saveName) then
					UI.saveLoadStatus.Text = "Editing: " .. saveName
					UI.saveLoadWindow.Visible = false
				end
			end)
			-- Load (green)
			local loadBtn = F.makeSaveActionBtn(row, ICONS.LoadSave, Color3.fromRGB(50, 180, 70), -86)
			loadBtn.MouseButton1Click:Connect(function()
				if F.loadSave(saveName) then
					UI.saveLoadStatus.Text = "Loaded: " .. saveName
				end
			end)
			-- Update (blue)
			local updBtn = F.makeSaveActionBtn(row, ICONS.UpdateSave, Color3.fromRGB(0, 170, 200), -56)
			updBtn.MouseButton1Click:Connect(function()
				local clones = F.captureProjectSnapshot()
				F.writeSave(saveName, clones)
				UI.saveLoadStatus.Text = "Updated: " .. saveName
				State.refreshSavesList()
			end)
			-- Delete (red)

					State.refreshSavesList()
				end
			end)
		end
	end

	end

function F.sanitizeFileName(raw)
	local s = tostring(raw or "")
	s = s:gsub("[%c\\/:*?\"<>|]", "")
	s = s:gsub("^%s+", "")
	s = s:gsub("%s+$", "")
	if s:lower():sub(-4) == ".lua" then
		s = s:sub(1, -5)
	end
	s = s:sub(1, 60)
	if s == "" then s = "Gui2Lua" end
	return s .. ".lua"
end

function F.downloadGui2Lua()
	local fileName = F.sanitizeFileName(UI.downloadNameBox.Text)
	local result = "Failed"
	local okGen, code = pcall(F.exportToLua, F.getGui2LuaOptions())
	if okGen and type(code) == "string" then
		if typeof(writefile) == "function" then
			local okWrite = pcall(writefile, fileName, code)
			if okWrite then result = "Saved!" end
		else
			result = "No writefile"
		end
	end
	UI.downloadWindow.Visible = false
	UI.downloadButton.Text = result
	task.delay(1.5, function()
		if UI.downloadButton and UI.downloadButton.Parent then
			UI.downloadButton.Text = "Download"
		end
	end)
end

-- Mobile State.clipboard has a size limit, so split code into chunks
State.MOBILE_CODE_CHUNK_LIMIT = 20000
State.codeChunks = {""}
State.codeChunkIndex = 1

function F.splitCodeIntoChunks(code, limit)
	if #code <= limit then
		return {code}
	end
	local chunks = {}
	local current = {}
	local currentLen = 0
	for line in (code .. "\n"):gmatch("([^\n]*)\n") do
		local lineLen = #line + 1
		if currentLen + lineLen > limit and #current > 0 then
			table.insert(chunks, table.concat(current, "\n"))
			current = {}
			currentLen = 0
		end
		table.insert(current, line)
		currentLen = currentLen + lineLen
	end
	if #current > 0 then
		table.insert(chunks, table.concat(current, "\n"))
	end
	return chunks
end

function F.showCodeChunk()
	UI.codeOutputBox.Text = State.codeChunks[State.codeChunkIndex] or ""
	UI.codePageLabel.Text = string.format("Parte %d/%d", State.codeChunkIndex, #State.codeChunks)
end

function F.updateCodeOutputView()
	if #State.codeChunks > 1 then
		UI.codePaginationRow.Visible = true
		UI.codeOutputScroll.Position = State.codeScrollPagedPos
		UI.codeOutputScroll.Size = State.codeScrollPagedSize
	else
		UI.codePaginationRow.Visible = false
		UI.codeOutputScroll.Position = State.codeScrollNormalPos
		UI.codeOutputScroll.Size = State.codeScrollNormalSize
	end
	F.showCodeChunk()
end

function F.sortGradientKeypoints()
	table.sort(State.gradientKeypoints, function(a, b) return a.Time < b.Time end)
	local n = #State.gradientKeypoints
	if n > 0 then
		State.gradientKeypoints[1].Time = 0
		State.gradientKeypoints[n].Time = 1
	end
	for i = 2, n - 1 do
		if State.gradientKeypoints[i].Time <= State.gradientKeypoints[i - 1].Time then
			State.gradientKeypoints[i].Time = State.gradientKeypoints[i - 1].Time + 0.0001
		end
	end
	if n >= 2 and State.gradientKeypoints[n - 1].Time >= State.gradientKeypoints[n].Time then
		State.gradientKeypoints[n - 1].Time = State.gradientKeypoints[n].Time - 0.0001
	end
end

function F.buildColorSequence()
	local seq = {}
	for _, kp in ipairs(State.gradientKeypoints) do
		table.insert(seq, ColorSequenceKeypoint.new(kp.Time, kp.Color))
	end
	return seq
end

function F.applyGradientToInstance()
	if not State.gradientEditorTarget then return end
	local seq = F.buildColorSequence()
	local ok = pcall(function()
		State.gradientEditorTarget.Color = ColorSequence.new(seq)
		UI.gradientBarPreview.Color = ColorSequence.new(seq)
	end)
	if not ok and #State.gradientKeypoints >= 2 then
		local fallback = ColorSequence.new(State.gradientKeypoints[1].Color, State.gradientKeypoints[#State.gradientKeypoints].Color)
		State.gradientEditorTarget.Color = fallback
		UI.gradientBarPreview.Color = fallback
	end
end

function F.refreshGradientMarkers()
	for _, marker in ipairs(State.gradientMarkers) do
		marker:Destroy()
	end
	State.gradientMarkers = {}

	for _, kp in ipairs(State.gradientKeypoints) do
		local marker = F.new("ImageButton", {
			Name = "Marker",
			Size = UDim2.new(0, 16, 0, 16),
			AnchorPoint = Vector2.new(0.5, 0),
			Position = UDim2.new(kp.Time, 0, 0, 0),
			BackgroundTransparency = 1,
			Image = State.GRADIENT_MARKER_IMAGE,
			ImageColor3 = (kp == State.selectedKeypoint) and Color3.fromRGB(255, 221, 84) or Color3.fromRGB(190, 190, 190),
			ZIndex = 20,
			Parent = UI.gradientMarkerHolder,
		})

		marker.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				State.selectedKeypoint = kp
				State.draggingKeypoint = kp
				UI.gradientTimeBox.Text = tostring(F.round(kp.Time, 3))
				UI.gradientColorSwatch.BackgroundColor3 = kp.Color
				F.refreshGradientMarkers()
			end
		end)

		table.insert(State.gradientMarkers, marker)
	end
end

function F.exportToLua(options)
	options = options or {}
	local lines = {}
	local varCount = 0
	local G2L = {}

	local function nextKey(className)
		varCount = varCount + 1
		local safe = className:gsub("[^%w_]", "")
		if safe == "" then safe = "Obj" end
		return safe .. "_" .. varCount
	end

	-- Header style @Uniquadev / Gui2Lua
	table.insert(lines, "--[=[")
	table.insert(lines, " d888b  db    db d888888b      .d888b.      db      db    db  .d8b.  ")
	table.insert(lines, "88' Y8b 88    88   `88'        VP  `8D      88      88    88 d8' `8b ")
	table.insert(lines, "88      88    88    88            odD'      88      88    88 88ooo88 ")
	table.insert(lines, "88  ooo 88    88    88          .88'        88      88    88 88~~~88 ")
	table.insert(lines, "88. ~8~ 88b  d88   .88.        j88.         88booo. 88b  d88 88   88    @uniquadev")
	table.insert(lines, " Y888P  ~Y8888P' Y888888P      888888D      Y88888P ~Y8888P' YP   YP  CONVERTER ")
	table.insert(lines, "")
	table.insert(lines, "designed using GuiBuilderTool - Enhanced Gui2Lua style")
	table.insert(lines, "]=]")
	table.insert(lines, "")
	table.insert(lines, "-- Services")
	table.insert(lines, "local Players = game:GetService(\"Players\")")
	table.insert(lines, "")
	table.insert(lines, "local G2L = {}")
	table.insert(lines, "")

	local function exportInstance(inst, parentKey)
		if F.isOverlay(inst) then return end
		local className = inst.ClassName
		local props = EXPORTABLE_PROPS[className]
		local key = nextKey(className)

		if options.comments then
			table.insert(lines, string.format("-- %s : %q", className, inst.Name))
		end

		if parentKey then
			table.insert(lines, string.format("G2L[%q] = Instance.new(%q, G2L[%q])", key, className, parentKey))
		else
			table.insert(lines, string.format("G2L[%q] = Instance.new(%q)", key, className))
		end

		-- Always set Name
		table.insert(lines, string.format("G2L[%q][\"Name\"] = %q", key, inst.Name))

		if props then
			for _, propName in ipairs(props) do
				if propName == "Name" then
					-- already set
				else
					local ok, value = pcall(function() return inst[propName] end)
					if ok and value ~= nil then
						-- Skip default-ish Source empty? still export
						if propName == "Size" and options.autoSizeScale and inst:IsA("GuiObject") then
							local parentObj = inst.Parent
							local parentSize = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsoluteSize or inst.AbsoluteSize
							local sx = parentSize.X > 0 and (inst.AbsoluteSize.X / parentSize.X) or 0
							local sy = parentSize.Y > 0 and (inst.AbsoluteSize.Y / parentSize.Y) or 0
							table.insert(lines, string.format("G2L[%q][\"Size\"] = UDim2.new(%s, 0, %s, 0)", key, F.round(sx, 4), F.round(sy, 4)))
						elseif propName == "Position" and options.autoPositionScale and inst:IsA("GuiObject") then
							local parentObj = inst.Parent
							local parentSize = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsoluteSize or inst.AbsoluteSize
							local parentPos = (parentObj and parentObj:IsA("GuiObject")) and parentObj.AbsolutePosition or Vector2.new(0, 0)
							local relX = inst.AbsolutePosition.X - parentPos.X
							local relY = inst.AbsolutePosition.Y - parentPos.Y
							local px = parentSize.X > 0 and (relX / parentSize.X) or 0
							local py = parentSize.Y > 0 and (relY / parentSize.Y) or 0
							table.insert(lines, string.format("G2L[%q][\"Position\"] = UDim2.new(%s, 0, %s, 0)", key, F.round(px, 4), F.round(py, 4)))
						elseif propName == "Source" then
							-- multiline string for scripts
							local src = tostring(value)
							table.insert(lines, string.format("G2L[%q][\"Source\"] = %s", key, string.format("%q", src)))
						else
							table.insert(lines, string.format("G2L[%q][%q] = %s", key, propName, F.luaValue(value)))
						end
					end
				end
			end
		end

		-- Parent already set via Instance.new second arg for children; root needs Parent
		if not parentKey then
			if className == "ScreenGui" then
				table.insert(lines, string.format("G2L[%q].Parent = Players.LocalPlayer:WaitForChild(\"PlayerGui\")", key))
			end
		end

		if options.autoAddConstraint and inst:IsA("GuiObject") and inst.AbsoluteSize.Y > 0 then
			local ratio = inst.AbsoluteSize.X / inst.AbsoluteSize.Y
			local cKey = key .. "_Ratio"
			table.insert(lines, string.format("G2L[%q] = Instance.new(\"UIAspectRatioConstraint\", G2L[%q])", cKey, key))
			table.insert(lines, string.format("G2L[%q][\"AspectRatio\"] = %s", cKey, F.round(ratio, 3)))
		end

		table.insert(lines, "")
		for _, child in ipairs(inst:GetChildren()) do
			exportInstance(child, key)
		end
		return key

function F.makeHorizontalDraggable(frame)
	local dragging = false
	local dragInput, dragStart, startPos

	local function update(input)
		if not dragging then return end
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset)
	end

	frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			update(input)
		end
	end)
end

-- UIDragDetector options: Joystick / No Joystick + drag speed (smoothness)
State.DRAG_JOYSTICK_ATTR = "DragJoystickMode" -- "Joystick" (default) or "No Joystick"
State.DRAG_SPEED_ATTR = "DragSpeed" -- 0.02 to 1 (1 = normal, lower = slower and smoother)
State.dragHooks = setmetatable({}, { __mode = "k" })
State.joystickHolds = 0
State.joystickPrev = {}

function F.findThumbstickFrames()
	local list = {}
	local touchGui = PlayerGui:FindFirstChild("TouchGui")
	if touchGui then
		for _, d in ipairs(touchGui:GetDescendants()) do
			if d:IsA("Frame") and (d.Name == "DynamicThumbstickFrame" or d.Name == "ThumbstickFrame") then
				table.insert(list, d)
			end
		end
	end
	return list
end

-- Hides only the thumbstick ball that follows the touch (visual only; walking input is not touched).
-- Returns a release function that restores it.
function F.holdThumbstickHidden()
	State.joystickHolds = State.joystickHolds + 1
	if State.joystickHolds == 1 then
		State.joystickPrev = {}
		local function apply()
			for _, f in ipairs(F.findThumbstickFrames()) do
				if State.joystickPrev[f] == nil then
					State.joystickPrev[f] = f.Visible
				end
				f.Visible = false
			end
		end
		apply()
		State.joystickConn = RunService.RenderStepped:Connect(apply)
	end
	local released = false
	return function()
		if released then return end
		released = true
		State.joystickHolds = math.max(0, State.joystickHolds - 1)
		if State.joystickHolds == 0 then
			if State.joystickConn then
				State.joystickConn:Disconnect()
				State.joystickConn = nil
			end
			for f, prev in pairs(State.joystickPrev) do
				pcall(function()
					if f.Parent then f.Visible = prev end
				end)
			end
			State.joystickPrev = {}
		end
	end
end

function F.hookDragDetector(det)
	if State.dragHooks[det] or not det:IsA("UIDragDetector") then return end
	local conns = {}
	State.dragHooks[det] = conns

	local dragging = false
	local hasBase = false
	local smoothed, target = UDim2.new(), UDim2.new()
	local loopConn = nil
	local releaseHide, watchdog = nil, nil

	local function endHide()
		if watchdog then
			watchdog:Disconnect()
			watchdog = nil
		end
		if releaseHide then
			local fn = releaseHide
			releaseHide = nil
			fn()
		end
	end

	local function beginHide()
		if releaseHide then return end
		releaseHide = F.holdThumbstickHidden()
		-- safety: never leave the joystick hidden if the drag is interrupted
		watchdog = RunService.Heartbeat:Connect(function()
			if not det.Parent or not det.Enabled then
				endHide()
			end
		end)
	end

	local function stopLoop()
		if loopConn then
			loopConn:Disconnect()
			loopConn = nil
		end
	end

	-- One smoothing step per rendered frame, so it does not depend on how often
	-- input events arrive. The object moves by the same amount "smoothed" moves.
	local function stepFrame(dt)
		local parent = det.Parent
		local speed = det:GetAttribute(State.DRAG_SPEED_ATTR)
		if not hasBase or not parent or not parent:IsA("GuiObject") or type(speed) ~= "number" or speed >= 1 then
			if not dragging then stopLoop() end
			return
		end
		speed = math.max(speed, 0.02)
		local alpha = 1 - (1 - speed) ^ (math.min(dt or (1 / 60), 0.1) * 60)
		local nextValue = smoothed:Lerp(target, alpha)
		local closeEnough = math.abs(target.X.Offset - nextValue.X.Offset) < 0.5
			and math.abs(target.Y.Offset - nextValue.Y.Offset) < 0.5
			and math.abs(target.X.Scale - nextValue.X.Scale) < 0.0005
			and math.abs(target.Y.Scale - nextValue.Y.Scale) < 0.0005
		local finished = closeEnough and not dragging
		if finished then
			nextValue = target
		end
		parent.Position = parent.Position + (nextValue - smoothed)
		smoothed = nextValue
		if finished then
			hasBase = false
			stopLoop()
		end
	end

	table.insert(conns, det.DragStart:Connect(function()
		stopLoop()
		dragging = true
		hasBase = false
		if det:GetAttribute(State.DRAG_JOYSTICK_ATTR) == "No Joystick" then
			beginHide()
		end
	end))

	table.insert(conns, det.DragEnd:Connect(function()
		dragging = false
		endHide()
		-- no snap on release: if it is still catching up, the loop keeps easing
		-- toward the release point at the same speed and stops by itself
		if not hasBase then
			stopLoop()
		end
	end))

	pcall(function()
		table.insert(conns, det:AddConstraintFunction(1000, function(pos, rot)
			local speed = det:GetAttribute(State.DRAG_SPEED_ATTR)
			if not dragging or type(speed) ~= "number" or speed >= 1 then
				return pos, rot
			end
			if not hasBase then
				-- the first proposal of the drag is the starting point (no jump)
				hasBase = true
				smoothed, target = pos, pos
				if not loopConn then
					loopConn = RunService.RenderStepped:Connect(stepFrame)
				end
				return pos, rot
			end
			target = pos
			return smoothed, rot
		end))
	end)

	table.insert(conns, det.Destroying:Connect(function()
		stopLoop()
		endHide()
		for _, c in ipairs(conns) do
			pcall(function() c:Disconnect() end)
		end
		State.dragHooks[det] = nil
	end))
end

function F.refreshProperties()
	for _, child in ipairs(UI.propertiesList:GetChildren()) do
		if child:IsA("GuiObject") then
			child:Destroy()
		end
	end
	if not State.selected then return end

	UI.propertiesTitle.Text = string.format('properties - %s "%s"',
		CLASS_LABELS_PT[State.selected.ClassName] or State.selected.ClassName, State.selected.Name)

	local searchText = UI.propertiesSearch.Text:lower()
	local order = 0
	for _, entry in ipairs(PROPERTY_SCHEMA) do
		if searchText == "" or entry.Label:lower():find(searchText, 1, true) then
			local row = F.buildPropertyRow(entry, order + 1)
			if row then
				order = order + 1
				row.Parent = UI.propertiesList
			end
		end
	end

	-- Corner control (round/rectangular), ScrollingFrame only
		if State.selected.ClassName == "ScrollingFrame" and (searchText == "" or ("canto"):find(searchText, 1, true)) then
		order = order + 1
		local cornerRow = F.new("Frame", {
			Size = UDim2.new(1, 0, 0, 20),
			BackgroundTransparency = 1,
			LayoutOrder = order,
			Parent = UI.propertiesList,
		})
		F.new("TextLabel", {
			Size = UDim2.new(0.46, 0, 1, 0),
			BackgroundTransparency = 1,
			Font = Enum.Font.Gotham,
			Text = "Corner",
			TextColor3 = THEME.SubText,
			TextSize = 11,
			TextXAlignment = Enum.TextXAlignment.Left,
			Parent = cornerRow,
		})

		local existingCorner = State.selected:FindFirstChildOfClass("UICorner")
		local cornerRadiusBox
		local cornerToggle = F.new("TextButton", {
			Size = UDim2.new(0.28, -2, 1, 0),
			Position = UDim2.new(0.46, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			Font = Enum.Font.Gotham,
			Text = existingCorner and "Arredondado" or "Retangular",
			TextColor3 = THEME.Text,
			TextSize = 10,
			Parent = cornerRow,
		})
		cornerRadiusBox = F.new("TextBox", {
			Size = UDim2.new(0.26, 0, 1, 0),
			Position = UDim2.new(0.74, 0, 0, 0),
			BackgroundColor3 = THEME.PanelAlt,
			ClearTextOnFocus = false,
			Font = Enum.Font.Gotham,
			Text = existingCorner and tostring(existingCorner.CornerRadius.Offset) or "0",
			TextColor3 = THEME.Text,
			TextSize = 10,
			Visible = existingCorner ~= nil,
			Parent = cornerRow,
		})

		cornerToggle.MouseButton1Click:Connect(function()
			local current = State.selected:FindFirstChildOfClass("UICorner")
			if current then
				current:Destroy()
				cornerToggle.Text = "Rectangular"
				cornerRadiusBox.Visible = false
			else
				F.new("UICorner", {
					CornerRadius = UDim.new(0, 8),
					Parent = State.selected,
				})
				cornerToggle.Text = "Rounded"
				cornerRadiusBox.Text = "8"
				cornerRadiusBox.Visible = true
			end
			F.refreshExplorer()
		end)

		cornerRadiusBox.FocusLost:Connect(function()
			local n = tonumber(cornerRadiusBox.Text)
			local current = State.selected:FindFirstChildOfClass("UICorner")
			if n and current then
				current.CornerRadius = UDim.new(0, n)
			end
		end)
	end

	-- UIDragDetector: Joystick dropdown + speed/smoothness
	if State.selected.ClassName == "UIDragDetector" then
		local det = State.selected

		if searchText == "" or ("joystick"):find(searchText, 1, true) then
			order = order + 1
			local joyRow = F.new("Frame", {
				Size = UDim2.new(1, 0, 0, 22),
				BackgroundTransparency = 1,
				LayoutOrder = order,
				Parent = UI.propertiesList,
			})
			F.new("TextLabel", {
				Size = UDim2.new(0.46, 0, 1, 0),
				BackgroundTransparency = 1,
				Font = Enum.Font.Gotham,
				Text = "Joystick",
				TextColor3 = THEME.SubText,
				TextSize = 11,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
				Parent = joyRow,
			})
			local joyMode = det:GetAttribute(State.DRAG_JOYSTICK_ATTR)
			if joyMode ~= "No Joystick" then joyMode = "Joystick" end
			local joyBox = F.new("TextButton", {
				Size = UDim2.new(0.54, 0, 1, 0),
				Position = UDim2.new(0.46, 0, 0, 0),
				BackgroundColor3 = THEME.PanelAlt,
				Font = Enum.Font.Gotham,
				Text = joyMode,
				TextColor3 = THEME.Text,
				TextSize = 11,
				TextXAlignment = Enum.TextXAlignment.Left,
				Parent = joyRow,
			})
			F.new("UIPadding", { PaddingLeft = UDim.new(0, 4), Parent = joyBox })
			F.attachDropdownArrow(joyBox)
			joyBox.MouseButton1Click:Connect(function()
				local items = {
					{ Label = "Joystick", Value = "Joystick" },
					{ Label = "No Joystick", Value = "No Joystick" },
				}
				F.openAnimatedDropdown(joyBox, items, joyBox.Text, function(picked)
					local value = type(picked) == "table" and picked.Value or picked

					F.hookDragDetector(det)
				end, { maxVisible = 2, width = math.max(joyBox.AbsoluteSize.X, 120) })

		end

		if searchText == "" or ("speed smooth drag"):find(searchText, 1, true) then
			order = order + 1
			local speedRow = F.new("Frame", {
				Size = UDim2.new(1, 0, 0, 22),

				TextSize = 11,

				TextTruncate = Enum.TextTruncate.AtEnd,
				Parent = speedRow,
			})
			local savedSpeed = det:GetAttribute(State.DRAG_SPEED_ATTR)
			if type(savedSpeed) ~= "number" then savedSpeed = 1 end
			local speedBox = F.new("TextBox", {
				Size = UDim2.new(0.54, 0, 1, 0),
				Position = UDim2.new(0.46, 0, 0, 0),
				BackgroundColor3 = THEME.PanelAlt,
				ClearTextOnFocus = false,
				Font = Enum.Font.Gotham,
				Text = tostring(F.round(savedSpeed, 3)),
				PlaceholderText = "0.02 - 1",
				TextColor3 = THEME.Text,
				TextSize = 11,
				Parent = speedRow,
			})
			speedBox.FocusLost:Connect(function()
				local n = tonumber(speedBox.Text)
				if n then
					n = math.clamp(n, 0.02, 1)
					det:SetAttribute(State.DRAG_SPEED_ATTR, n)
					F.hookDragDetector(det)
					speedBox.Text = tostring(F.round(n, 3))
				else
					local current = det:GetAttribute(State.DRAG_SPEED_ATTR)
					speedBox.Text = tostring(F.round(type(current) == "number" and current or 1, 3))
				end
			end)
		end
	end
end

-- Explorer Tree
function F.clearExplorer()
	for _, row in pairs(State.explorerRows) do
		if row.Row then row.Row:Destroy() end
	end
	State.explorerRows = {}
end

State.TREE_LINE_COLOR = Color3.fromRGB(255, 140, 0)
State.TREE_STEP_PX = 14

function F.buildExplorer(inst, depth, order, ancestorIsLast)
	if F.isOverlay(inst) then return order end
	ancestorIsLast = ancestorIsLast or {}

	order = order + 1
	local row = F.new("Frame", {
		Name = "Row",
		Size = UDim2.new(1, 0, 0, 24),
		BackgroundColor3 = THEME.Panel,
		BackgroundTransparency = getTargetPanelTransparency and F.getTargetPanelTransparency() or 0,
		LayoutOrder = order,
		Parent = UI.explorerList,
	})

	local indentPx = depth * State.TREE_STEP_PX

	-- Thin tree connector lines (parent -> child)
			if depth > 0 then
		for level = 1, depth - 1 do
			if not ancestorIsLast[level] then
				local colX = (level - 1) * State.TREE_STEP_PX + 7
				F.new("Frame", {
					Name = "TreeTrunk",
					Size = UDim2.new(0, 1, 1, 0),
					Position = UDim2.new(0, colX, 0, 0),
					BackgroundColor3 = State.TREE_LINE_COLOR,
					BorderSizePixel = 0,
					ZIndex = 2,
					Parent = row,
				})
			end
		end

		local myColX = (depth - 1) * State.TREE_STEP_PX + 7
		local isLastHere = ancestorIsLast[depth]

		-- top half (always)
		F.new("Frame", {
			Name = "TreeBranchTop",
			Size = UDim2.new(0, 1, 0, 12),
			Position = UDim2.new(0, myColX, 0, 0),
			BackgroundColor3 = State.TREE_LINE_COLOR,
			BorderSizePixel = 0,
			ZIndex = 2,
			Parent = row,
		})
		-- bottom half (only if more siblings follow)
				if not isLastHere then
			F.new("Frame", {
				Name = "TreeBranchBottom",
				Size = UDim2.new(0, 1, 0, 12),
				Position = UDim2.new(0, myColX, 0, 12),
				BackgroundColor3 = State.TREE_LINE_COLOR,
				BorderSizePixel = 0,
				ZIndex = 2,
				Parent = row,
			})
		end
		-- horizontal branch to the label
		F.new("Frame", {
			Name = "TreeBranchH",
			Size = UDim2.new(0, indentPx - myColX, 0, 1),
			Position = UDim2.new(0, myColX, 0, 12),
			BackgroundColor3 = State.TREE_LINE_COLOR,
			BorderSizePixel = 0,
			ZIndex = 2,
			Parent = row,
		})
	end

	local selectButton = F.new("TextButton", {
		Size = UDim2.new(1, -28 - indentPx, 1, 0),
		Position = UDim2.new(0, indentPx, 0, 0),
		BackgroundColor3 = (inst == State.selected) and THEME.RowSelected or THEME.Panel,
		BackgroundTransparency = getTargetPanelTransparency and F.getTargetPanelTransparency() or 0,
		AutoButtonColor = false,
		Font = Enum.Font.Gotham,
		Text = (CLASS_LABELS_PT[inst.ClassName] or inst.ClassName) .. ' "' .. inst.Name .. '"',
		TextColor3 = THEME.Text,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = row,
	})
	selectButton.MouseButton1Click:Connect(function()
		F.selectInstance(inst)
	end)

	local plusButton = F.new("ImageButton", {
		Size = UDim2.new(0, 20, 0, 20),
		Position = UDim2.new(1, -24, 0, 2),
		BackgroundColor3 = THEME.Button,
		Image = ICONS.Plus,
		ScaleType = Enum.ScaleType.Fit,
		ZIndex = 2,
		Parent = row,
	})
	F.new("UICorner", { CornerRadius = UDim.new(0, 4), Parent = plusButton })
	plusButton.MouseButton1Click:Connect(function()
		F.openAddElementPopup(inst, row)
	end)

	State.explorerRows[#State.explorerRows + 1] = { Row = row, Button = selectButton, Instance = inst }

	local visibleChildren = {}
	for _, child in ipairs(inst:GetChildren()) do
		if not F.isOverlay(child) then
			table.insert(visibleChildren, child)
		end
	end

	for i, child in ipairs(visibleChildren) do
		local childAncestorIsLast = {}
		for k, v in pairs(ancestorIsLast) do
			childAncestorIsLast[k] = v
		end
		childAncestorIsLast[depth + 1] = (i == #visibleChildren)
		order = F.buildExplorer(child, depth + 1, order, childAncestorIsLast)
	end
	return order
end

function F.refreshExplorer()
	F.clearExplorer()
	F.buildExplorer(State.projectGui, 0, 0)
end

-- Element Creation
function F.addElement(className, targetInstance)
	State.elementCounter = State.elementCounter + 1
	local defaults = CLASS_DEFAULTS[className] or {}
	local props = {}
	for k, v in pairs(defaults) do
		props[k] = v
	end
	props.Name = className .. State.elementCounter

	local parent = targetInstance or State.projectGui
	local inst = F.new(className, props)
	inst.Parent = parent

	if (className == "LocalScript" or className == "ModuleScript") and defaults.Source then
		inst.Source = defaults.Source
	end

	-- ScrollingFrame starts rounded (can go rectangular)
		if className == "ScrollingFrame" then
		F.new("UICorner", {
			CornerRadius = UDim.new(0, 8),
			Parent = inst,
		})
	end

	F.refreshExplorer()
	F.selectInstance(inst)
end

-- Theme
function F.applyTheme(themeName)
	if not THEMES[themeName] then return end
	State.currentThemeName = themeName
	for k, v in pairs(THEMES[themeName]) do
		THEME[k] = v
	end

	UI.explorerWindow.BackgroundColor3 = THEME.Background
	UI.propertiesWindow.BackgroundColor3 = THEME.Background
	UI.settingsWindow.BackgroundColor3 = THEME.Background
	UI.saveLoadWindow.BackgroundColor3 = THEME.Background
	UI.gui2luaWindow.BackgroundColor3 = THEME.Background
	if UI.downloadWindow then
		UI.downloadWindow.BackgroundColor3 = THEME.Background
		UI.downloadTop.BackgroundColor3 = THEME.Panel
		UI.downloadNameBox.BackgroundColor3 = THEME.PanelAlt
		UI.downloadNameBox.TextColor3 = THEME.Text
		UI.downloadNameBox.PlaceholderColor3 = THEME.SubText
	end
	UI.addElementPopup.BackgroundColor3 = THEME.Background
	UI.codeOutputWindow.BackgroundColor3 = THEME.Panel
	-- UI.optionsMenu color is fixed, not theme-based
	UI.sourceEditorWindow.BackgroundColor3 = THEME.Background

	UI.explorerTop.BackgroundColor3 = THEME.Panel
	UI.propertiesTop.BackgroundColor3 = THEME.Panel
	UI.settingsTop.BackgroundColor3 = THEME.Panel
	UI.saveLoadTop.BackgroundColor3 = THEME.Panel
	UI.gui2luaTop.BackgroundColor3 = THEME.Panel
	UI.addElementTop.BackgroundColor3 = THEME.Panel
	UI.codeOutputTop.BackgroundColor3 = THEME.PanelAlt
	UI.sourceEditorTop.BackgroundColor3 = THEME.Panel
	UI.actionBar.BackgroundColor3 = THEME.Panel

	for _, t in ipairs({UI.themeToggleBlack, UI.themeToggleWhite}) do
		local active = (t.themeName == State.currentThemeName)
		t.track.BackgroundColor3 = active and THEME.Accent or THEME.Button
		t.knob.Position = active and UDim2.new(1, -22, 0.5, -10) or UDim2.new(0, 2, 0.5, -10)
	end

	F.refreshExplorer()
	F.refreshProperties()
	if saveSettingsToDisk then F.saveSettingsToDisk() end
	if applyUiTransparency then F.applyUiTransparency(State.uiTransparencyEnabled) end
end

function F.refreshGuiLists()
	State.coreGuiList = { "none" }
	State.playerGuiList = { "none" }
	-- CoreGui (may fail without permission)
	pcall(function()
		local CoreGui = game:GetService("CoreGui")
		for _, child in ipairs(CoreGui:GetChildren()) do
			if child:IsA("ScreenGui") or child:IsA("LayerCollector") or child:IsA("Folder") then
				table.insert(State.coreGuiList, child.Name)
			end
		end
	end)
	for _, child in ipairs(PlayerGui:GetChildren()) do
		if child:IsA("ScreenGui") and child.Name ~= "GuiBuilderTool" then
			table.insert(State.playerGuiList, child.Name)
		end
	end
	UI.coreGuiDrop.Text = tostring(State.selectedCoreGuiName)
	UI.playerGuiDrop.Text = tostring(State.selectedPlayerGuiName)
end

UI.coreGuiDrop.MouseButton1Click:Connect(function()
	F.refreshGuiLists()
	local items = {}
	for _, name in ipairs(State.coreGuiList) do
		table.insert(items, { Label = name, Value = name })
	end
	F.openAnimatedDropdown(UI.coreGuiDrop, items, State.selectedCoreGuiName, function(picked)
		local name = type(picked) == "table" and picked.Value or picked
		State.selectedCoreGuiName = name
		UI.coreGuiDrop.Text = name
	end, { maxVisible = 7, width = math.max(UI.coreGuiDrop.AbsoluteSize.X, 200) })
end)

UI.playerGuiDrop.MouseButton1Click:Connect(function()
	F.refreshGuiLists()
	local items = {}
	for _, name in ipairs(State.playerGuiList) do
		table.insert(items, { Label = name, Value = name })
	end
	F.openAnimatedDropdown(UI.playerGuiDrop, items, State.selectedPlayerGuiName, function(picked)
		local name = type(picked) == "table" and picked.Value or picked
		State.selectedPlayerGuiName = name
		UI.playerGuiDrop.Text = name
	end, { maxVisible = 7, width = math.max(UI.playerGuiDrop.AbsoluteSize.X, 200) })
end)

function F.setProjectRoot(gui)
	if not gui then return end
	-- keep previous, just switch focus
	State.projectGui = gui
	State.selected = nil
	F.destroyBoundingBox()
	F.clearAlignmentLines()
	F.refreshExplorer()
	F.selectInstance(State.projectGui)
	UI.explorerWindow.Visible = true
	UI.propertiesWindow.Visible = true
end

UI.modGuiApplyBtn.MouseButton1Click:Connect(function()
	UI.modGuiWindow.Visible = false
	-- Player GUI takes priority over CoreGui
	if State.selectedPlayerGuiName and State.selectedPlayerGuiName ~= "none" then
		local gui = PlayerGui:FindFirstChild(State.selectedPlayerGuiName)
		if gui then
			F.setProjectRoot(gui)
			State.editingExternalGui = gui
		end
	elseif State.selectedCoreGuiName and State.selectedCoreGuiName ~= "none" then
		pcall(function()
			local CoreGui = game:GetService("CoreGui")
			local gui = CoreGui:FindFirstChild(State.selectedCoreGuiName)
			if gui then
				F.setProjectRoot(gui)
				State.editingExternalGui = gui
			end
		end)
	end
end)

UI.modGuiCloseBtn.MouseButton1Click:Connect(function()
	UI.modGuiWindow.Visible = false
end)

-- Wire mod gui toggle
if UI.modGuiClick then
	UI.modGuiClick.MouseButton1Click:Connect(function()
		State.modGuiEnabled = not State.modGuiEnabled
		UI.modGuiTrack.BackgroundColor3 = State.modGuiEnabled and THEME.Accent or THEME.Button
		UI.modGuiKnob.Position = State.modGuiEnabled and UDim2.new(1, -22, 0.5, -10) or UDim2.new(0, 2, 0.5, -10)
		if State.modGuiEnabled then
			F.refreshGuiLists()
			UI.modGuiWindow.Visible = true
		else
			UI.modGuiWindow.Visible = false
			if State.userProjectGui and State.userProjectGui.Parent then
				F.setProjectRoot(State.userProjectGui)
			end
		end
	end)
end
