local function makeIcon(parent, assetId, size, posX, posY, zIndex)
    local img = Instance.new("ImageLabel")
    img.Size                  = UDim2.new(0, size, 0, size)
    img.Position              = UDim2.new(0, posX, 0, posY)
    img.BackgroundTransparency = 1
    img.Image                 = assetId
    img.ScaleType             = Enum.ScaleType.Fit
    img.ZIndex                = zIndex or 2
    img.Parent                = parent
    return img
end

-- Helper: line with icon + TextLabel
local function makeIconRow(parent, assetId, text, posY, iconSize, textSize, zIndex)
    iconSize = iconSize or 16
    textSize = textSize or 12
    zIndex   = zIndex or 2

    local container = Instance.new("Frame")
    container.Size                  = UDim2.new(1, -14, 0, iconSize)
    container.Position              = UDim2.new(0, 10, 0, posY)
    container.BackgroundTransparency = 1
    container.ZIndex                = zIndex
    container.Parent                = parent

    makeIcon(container, assetId, iconSize, 0, 0, zIndex)

    local lbl = Instance.new("TextLabel")
    lbl.Size             = UDim2.new(1, -(iconSize + 6), 1, 0)
    lbl.Position         = UDim2.new(0, iconSize + 6, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text             = text
    lbl.TextColor3       = Color3.fromRGB(200, 200, 200)
    lbl.TextSize         = textSize
    lbl.Font             = Enum.Font.Gotham
    lbl.TextXAlignment   = Enum.TextXAlignment.Left
    lbl.TextTruncate     = Enum.TextTruncate.AtEnd
    lbl.ZIndex           = zIndex
    lbl.Parent           = container

    return container, lbl
end

--  DRAG
-- ══════════════════════════════════════════
local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

local function updateDrag(input)
    if not dragging then return end
    local delta = input.Position - dragStart
    TweenService:Create(MainFrame, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {
        Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    }):Play()
end

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then updateDrag(input) end
end)

local function removeCard(player)
    if playerCards[player.Name] then
        playerCards[player.Name].card:Destroy()
        playerCards[player.Name] = nil
    end
    playerData[player.Name] = nil
end

-- ══════════════════════════════════════════
--  DETECTION
-- ══════════════════════════════════════════
local function isGrounded(hrp, character)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { character }
    params.FilterType = Enum.RaycastFilterType.Exclude
    return workspace:Raycast(hrp.Position, Vector3.new(0, -4.5, 0), params) ~= nil
end

local function checkPlayer(player)
    local character = player.Character
    if not character then return end

    local hrp      = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not hrp or not humanoid then return end

    local card = playerCards[player.Name]
    if not card then return end

    if not playerData[player.Name] then
        playerData[player.Name] = { airTime = 0 }
    end
    local data = playerData[player.Name]

    local vel        = hrp.AssemblyLinearVelocity
    local hSpeed     = Vector3.new(vel.X, 0, vel.Z).Magnitude
    local totalSpeed = vel.Magnitude
    local walkSpeed  = math.max(humanoid.WalkSpeed, 1)
    local jumpPower  = humanoid.JumpPower

    local grounded = isGrounded(hrp, character)
    data.airTime   = grounded and 0 or (data.airTime + CONFIG.CHECK_INTERVAL)

    local flags = {}

    if hSpeed > walkSpeed * CONFIG.SPEED_MULTIPLIER and hSpeed > 5 then
        table.insert(flags, "Speed")
    end
    if data.airTime > CONFIG.MAX_AIR_TIME and vel.Y > -8 then
        table.insert(flags, "Fly")
    end
    if totalSpeed > CONFIG.FLING_THRESHOLD then
        table.insert(flags, "Fling")
    end
    if vel.Y > jumpPower * CONFIG.JUMP_MULTIPLIER and vel.Y > 10 then
        table.insert(flags, "Abnormal Jump")
    end

    card.studLbl.Text = string.format("Studs/s: %.1f  (WS:%d)", hSpeed, walkSpeed)

    if #flags > 0 then
        card.statusBar.BackgroundColor3 = COLOR_CHEAT
        card.statusIcon.Image           = ICONS.scripter
        card.statusLbl.Text             = "SCRIPTER DETECTED"
        card.statusLbl.TextColor3       = COLOR_CHEAT
        card.userLbl.TextColor3         = COLOR_CHEAT
        card.dispLbl.TextColor3         = COLOR_CHEAT
        card.flagsLbl.Text              = table.concat(flags, " | ")
        card.flagsLbl.TextColor3        = COLOR_CHEAT
    else
        card.statusBar.BackgroundColor3 = COLOR_CLEAN
        card.statusIcon.Image           = ICONS.normal
        card.statusLbl.Text             = "Normal"
        card.statusLbl.TextColor3       = COLOR_CLEAN
        card.userLbl.TextColor3         = COLOR_NEUTRAL
        card.dispLbl.TextColor3         = COLOR_NEUTRAL
        card.flagsLbl.Text              = ""
    end
end

-- ══════════════════════════════════════════
--  PLAYER SETUP
-- ══════════════════════════════════════════
local function setupPlayer(player)
    createCard(player)
    player.CharacterAdded:Connect(function()
        playerData[player.Name] = nil
    end)
end

for _, player in ipairs(Players:GetPlayers()) do
    setupPlayer(player)
end

Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(removeCard)

-- ══════════════════════════════════════════
--  MAIN LOOP
-- ══════════════════════════════════════════
local elapsed = 0
RunService.Heartbeat:Connect(function(dt)
    if not detectorEnabled then return end
    elapsed += dt
    if elapsed < CONFIG.CHECK_INTERVAL then return end
    elapsed = 0
    for _, player in ipairs(Players:GetPlayers()) do
        pcall(checkPlayer, player)
    end
end)
